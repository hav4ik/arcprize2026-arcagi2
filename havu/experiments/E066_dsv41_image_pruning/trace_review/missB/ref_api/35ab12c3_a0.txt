from itertools import combinations, permutations

def aligned(p,q):
    dr=q[0]-p[0]; dc=q[1]-p[1]
    return dr==0 or dc==0 or abs(dr)==abs(dc)

def line_cells(p,q):
    dr=q[0]-p[0]; dc=q[1]-p[1]
    n=max(abs(dr),abs(dc))
    sr=0 if dr==0 else (1 if dr>0 else -1)
    sc=0 if dc==0 else (1 if dc>0 else -1)
    return [(p[0]+sr*k,p[1]+sc*k) for k in range(n+1)]

def build_shapes(g):
    H,W=g.shape
    colors=[c for c in range(1,10) if np.any(g==c)]
    seeds={c:list(zip(*np.where(g==c))) for c in colors}
    shape_cells={}  # color -> set of cells drawn
    for c in colors:
        pts=seeds[c]
        if len(pts)<2:
            continue
        edges=[(i,j) for i in range(len(pts)) for j in range(i+1,len(pts)) if aligned(pts[i],pts[j])]
        best=None
        m=len(edges)
        for mask in range(1<<m):
            deg=[0]*len(pts); ec=0; tot=0; ok=True
            for k,(i,j) in enumerate(edges):
                if mask>>k &1:
                    deg[i]+=1; deg[j]+=1
                    if deg[i]>2 or deg[j]>2:
                        ok=False;break
                    ec+=1
                    dr=abs(pts[i][0]-pts[j][0]); dc=abs(pts[i][1]-pts[j][1])
                    tot+=max(dr,dc)
            if not ok: continue
            key=(-ec, tot)
            if best is None or key<best[0]:
                best=(key,mask)
        mask=best[1] if best else 0
        cells=set(pts)
        for k,(i,j) in enumerate(edges):
            if mask>>k &1:
                cells.update(line_cells(pts[i],pts[j]))
        shape_cells[c]=cells
    # copies for single-seed colors
    base_colors=[c for c in colors if len(seeds[c])>=2]
    copies={}
    for c in colors:
        if len(seeds[c])==1:
            p=seeds[c][0]
            cand=None
            for b in base_colors:
                for v in seeds[b]:
                    d=max(abs(p[0]-v[0]),abs(p[1]-v[1]))
                    if d==1:
                        off=(p[0]-v[0],p[1]-v[1])
                        if cand is None:
                            cand=(b,off)
            copies[c]=cand
    return seeds,shape_cells,copies

def solve(g):
    g=np.array(g)
    seeds,shape_cells,copies=build_shapes(g)
    out=np.zeros_like(g)
    # bases first
    for c,cells in shape_cells.items():
        for (r,cc) in cells:
            if 0<=r<out.shape[0] and 0<=cc<out.shape[1]:
                out[r,cc]=c
    # copies
    for c,cand in copies.items():
        if cand is None: 
            p=seeds[c][0]; out[p]=c; continue
        b,off=cand
        for (r,cc) in shape_cells[b]:
            nr,nc=r+off[0],cc+off[1]
            if 0<=nr<out.shape[0] and 0<=nc<out.shape[1] and out[nr,nc]==0:
                out[nr,nc]=c
    return out
