import numpy as np
from scipy.ndimage import label, find_objects

def comp_rotate_grid(sub,k,roundmode='round'):
    out=np.zeros_like(sub)
    for col in sorted(set(sub.ravel())-{0}):
        lab,n=label(sub==col, structure=np.ones((3,3)))
        for i,sl in enumerate(find_objects(lab),1):
            if sl is None: continue
            y0,y1=sl[0].start,sl[0].stop; x0,x1=sl[1].start,sl[1].stop
            m=(lab[sl]==i)
            t=np.rot90(m,k)
            cy=(y0+y1-1)/2; cx=(x0+x1-1)/2
            ty=cy-(t.shape[0]-1)/2; tx=cx-(t.shape[1]-1)/2
            if roundmode=='round': ty=int(np.floor(ty+0.5)); tx=int(np.floor(tx+0.5))
            elif roundmode=='floor': ty=int(np.floor(ty)); tx=int(np.floor(tx))
            elif roundmode=='ceil': ty=int(np.ceil(ty)); tx=int(np.ceil(tx))
            yy,xx=np.where(t)
            out[yy+ty,xx+tx]=col
    return out
