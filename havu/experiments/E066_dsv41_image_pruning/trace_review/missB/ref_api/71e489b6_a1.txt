from collections import deque

def comps(a, val):
    H,W=a.shape
    seen=np.zeros_like(a,bool); res=[]
    for r in range(H):
        for c in range(W):
            if a[r,c]==val and not seen[r,c]:
                q=[(r,c)]; seen[r,c]=True; cells=[]
                while q:
                    x,y=q.pop(); cells.append((x,y))
                    for dx,dy in [(1,0),(-1,0),(0,1),(0,-1)]:
                        nx,ny=x+dx,y+dy
                        if 0<=nx<H and 0<=ny<W and a[nx,ny]==val and not seen[nx,ny]:
                            seen[nx,ny]=True; q.append((nx,ny))
                res.append(cells)
    return res
