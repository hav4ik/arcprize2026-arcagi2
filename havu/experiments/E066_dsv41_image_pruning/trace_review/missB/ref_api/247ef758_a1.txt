from collections import Counter

def solve(g):
    g=np.array(g)
    H,W=g.shape
    # find separator column: full column single nonzero color, with frame to right
    s=None
    for j in range(1,W-2):
        col=g[:,j]
        if col.min()==col.max() and col[0]!=0 and j+2<=W-1:
            # require left pane nonempty and a border to right
            s=j; break
    # frame border cols s+1..W-1
    lb=s+1; rb=W-1
    border=[]
    border+=list(g[0,lb:rb+1])+list(g[H-1,lb:rb+1])+list(g[1:H-1,lb])+list(g[1:H-1,rb])
    dom=Counter(border).most_common(1)[0][0]
    topcols={}
    for c in range(lb+1,rb):
        v=g[0,c]
        if v!=dom and v!=0:
            topcols.setdefault(int(v),set()).add(c)
    siderows={}
    for r in range(1,H-1):
        v=g[r,lb]
        if v!=dom and v!=0:
            siderows.setdefault(int(v),set()).add(r)
    # also right side
    for r in range(1,H-1):
        v=g[r,rb]
        if v!=dom and v!=0:
            siderows.setdefault(int(v),set()).add(r)
    # left pane shapes
    left=g[:,:s]
    colors=sorted(set(int(x) for x in np.unique(left) if x!=0))
    shapes=[]
    for col in colors:
        rs,cs=np.where(left==col)
        rmin,rmax,cmin,cmax=rs.min(),rs.max(),cs.min(),cs.max()
        cr=(rmin+rmax)/2; cc=(cmin+cmax)/2
        offs=[(r-cr,c-cc) for r,c in zip(rs,cs)]
        shapes.append((rmin,col,offs,cr,cc))
    shapes.sort(key=lambda t:t[0])
    out=g.copy()
    out[:,:s]=0
    for rmin,col,offs,cr,cc in shapes:
        if col not in topcols or col not in siderows: continue
        for r in siderows[col]:
            for c in topcols[col]:
                for dr,dc in offs:
                    rr=int(round(r+dr)); ccv=int(round(c+dc))
                    if 0<=rr<H and 0<=ccv<W:
                        if out[rr,ccv]==0:
                            out[rr,ccv]=col
    return out

def solve(g):
    g=np.array(g)
    H,W=g.shape
    s=None
    for j in range(1,W-2):
        col=g[:,j]
        if col.min()==col.max() and col[0]!=0:
            s=j; break
    lb=s+1; rb=W-1
    border=list(g[0,lb:rb+1])+list(g[H-1,lb:rb+1])+list(g[1:H-1,lb])+list(g[1:H-1,rb])
    dom=Counter(border).most_common(1)[0][0]
    topcols={}
    for c in range(lb+1,rb):
        v=int(g[0,c])
        if v!=dom and v!=0: topcols.setdefault(v,set()).add(c)
    siderows={}
    for r in range(1,H-1):
        for cc in (lb,rb):
            v=int(g[r,cc])
            if v!=dom and v!=0: siderows.setdefault(v,set()).add(r)
    left=g[:,:s]
    colors=sorted(int(x) for x in np.unique(left) if x!=0)
    shapes=[]
    for col in colors:
        rs,cs=np.where(left==col)
        cr=(rs.min()+rs.max())/2; cc=(cs.min()+cs.max())/2
        offs=[(r-cr,c-cc) for r,c in zip(rs,cs)]
        shapes.append((rs.min(),col,rs,cs,offs))
    shapes.sort(key=lambda t:t[0])
    out=g.copy()
    for rmin,col,rs,cs,offs in shapes:
        if col in topcols and col in siderows:
            out[rs,cs]=0
            for r in siderows[col]:
                for c in topcols[col]:
                    for dr,dc in offs:
                        rr=int(round(r+dr)); ccv=int(round(c+dc))
                        if 0<=rr<H and 0<=ccv<W and out[rr,ccv]==0:
                            out[rr,ccv]=col
    return out

def solve2(g):
    g=np.array(g)
    H,W=g.shape
    s=None
    for j in range(1,W-2):
        col=g[:,j]
        if col.min()==col.max() and col[0]!=0:
            s=j; break
    lb=s+1; rb=W-1
    border=list(g[0,lb:rb+1])+list(g[H-1,lb:rb+1])+list(g[1:H-1,lb])+list(g[1:H-1,rb])
    dom=Counter(border).most_common(1)[0][0]
    topcols={}
    for c in range(lb+1,rb):
        v=int(g[0,c])
        if v!=dom and v!=0: topcols.setdefault(v,set()).add(c)
    siderows={}
    for r in range(1,H-1):
        for cc in (lb,rb):
            v=int(g[r,cc])
            if v!=dom and v!=0: siderows.setdefault(v,set()).add(r)
    left=g[:,:s]
    colors=sorted(int(x) for x in np.unique(left) if x!=0)
    shapes={}
    for col in colors:
        rs,cs=np.where(left==col)
        cr=(rs.min()+rs.max())/2; cc=(cs.min()+cs.max())/2
        shapes[col]=([(r-cr,c-cc) for r,c in zip(rs,cs)],rs,cs)
    placements=[]
    for col,(offs,rs,cs) in shapes.items():
        if col in topcols and col in siderows:
            for r in siderows[col]:
                for c in topcols[col]:
                    placements.append((r,c,col,offs,rs,cs))
    placements.sort(key=lambda t:(t[0],t[1]))
    out=g.copy()
    for col,(offs,rs,cs) in shapes.items():
        if col in topcols and col in siderows:
            out[rs,cs]=0
    for r,c,col,offs,rs,cs in placements:
        for dr,dc in offs:
            rr=int(round(r+dr)); ccv=int(round(c+dc))
            if 0<=rr<H and 0<=ccv<W and out[rr,ccv]==0:
                out[rr,ccv]=col
    return out
