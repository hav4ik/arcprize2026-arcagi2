import numpy as np

def orth_counts(inp):
    H,W=inp.shape
    b=np.zeros((H,W),int)
    for r in range(H):
      for c in range(W):
        b[r,c]=sum(0<=r+dr<H and 0<=c+dc<W and inp[r+dr,c+dc]==1 for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)])
    return b

def solve_cand(inp, add_rule='two'):
    inp=np.array(inp); H,W=inp.shape
    b=orth_counts(inp)
    # erase blue noise
    out=inp.copy()
    for r in range(H):
      for c in range(W):
        if inp[r,c]==1 and b[r,c]<=1:
          out[r,c]=0
    # seeds
    M=set()
    for r in range(H):
      for c in range(W):
        if inp[r,c]==0:
          if b[r,c]>=3:
            M.add((r,c))
          elif (r in (0,H-1) or c in (0,W-1)) and b[r,c]>=2:
            M.add((r,c))
    # iterative
    changed=True
    while changed:
      changed=False
      for r in range(H):
        for c in range(W):
          if inp[r,c]==0 and (r,c) not in M and b[r,c]>=2:
            nb=[(r+dr,c+dc) for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)] if 0<=r+dr<H and 0<=c+dc<W]
            marked_nb=sum(p in M for p in nb)
            if add_rule=='two' and marked_nb>=2:
              M.add((r,c)); changed=True
            elif add_rule=='all' and all(inp[p]==0 and p in M for p in nb):
              M.add((r,c)); changed=True
    # paint
    for r,c in M:
      for dr in [-1,0,1]:
        for dc in [-1,0,1]:
          rr,cc=r+dr,c+dc
          if 0<=rr<H and 0<=cc<W: out[rr,cc]=7
    for r,c in M: out[r,c]=0
    return out,M

def solve_final(inp):

    # blue neighbor count in original grid

    dirs=[(-1,0),(1,0),(0,-1),(0,1)]
    for r in range(H):
        for c in range(W):
            b[r,c]=sum(0<=r+dr<H and 0<=c+dc<W and inp[r+dr,c+dc]==1 for dr,dc in dirs)
    # erase thin blue protrusions/noise
    clean=inp.copy()
    for r in range(H):
        for c in range(W):
            if inp[r,c]==1 and b[r,c]<=1:

    # original black cells that are thin black protrusions/holes

    # fill compact black clusters attached to such cells
