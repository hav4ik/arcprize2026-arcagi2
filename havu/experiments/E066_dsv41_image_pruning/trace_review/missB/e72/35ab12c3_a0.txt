def line_cells(a,b):
    dr=b[0]-a[0]; dc=b[1]-a[1]
    if dr==0:
        return [(a[0],c) for c in range(min(a[1],b[1]),max(a[1],b[1])+1)]

        return [(r,a[1]) for r in range(min(a[0],b[0]),max(a[0],b[0])+1)]

    return []

def line_cells(a,b):
    dr=b[0]-a[0]; dc=b[1]-a[1]
    if dr==0:

        return [(x,a[1]) for x in range(min(a[0],b[0]),max(a[0],b[0])+1)]

        s=1 if dr>0 else -1; t=1 if dc>0 else -1
        return [(a[0]+s*k,a[1]+t*k) for k in range(abs(dr)+1)]
    return []
