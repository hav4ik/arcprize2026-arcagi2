    if dr==0 and dc==0: return [tuple(p)]

def paircells(p,q):
    dr=q[0]-p[0]; dc=q[1]-p[1]
    if dr==0 and dc==0: return [tuple(p)]
    if not (dr==0 or dc==0 or abs(dr)==abs(dc)): return []
    n=max(abs(dr),abs(dc))
    sr=0 if dr==0 else dr//n
    sc=0 if dc==0 else dc//n
    return [(p[0]+i*sr,p[1]+i*sc) for i in range(n+1)]
