def solve(t, mode='mincolor'):
    a,dc,base,left,right,colmarks,rowmarks,shapes=parse(t)
    out=a.copy()
    H,W=a.shape
    marked=[c for c in shapes if (c in colmarks and c in rowmarks)]
    # clear marked shapes in left
    for c in marked:
        for r,cc in shapes[c][0]:
            out[r,cc]=0
    # order
    if mode=='mincolor': order=sorted(marked, reverse=True) # larger first, min last wins
    elif mode=='maxcolor': order=sorted(marked)
    elif mode=='tops': order=sorted(marked, key=lambda c: min(colmarks[c]))
    elif mode=='rows': order=sorted(marked, key=lambda c: min(rowmarks[c]))
    elif mode=='leftpanel':
        # order by shape top row
        order=sorted(marked, key=lambda c: shapes[c][1][0])
    else:
        order=marked
    for c in order:
        cells,(r0,r1,c0,c1)=shapes[c]

            for ccol in colmarks[c]:
                for dr,dcx in offs:
                    R,C=rr+dr,ccol+dcx
                    out[R,C]=c
    return out
