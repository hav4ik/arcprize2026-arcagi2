import numpy as np

S8=np.ones((3,3),bool)

def get_templates(g,wall):
    left=g[:,:wall]
    lab,n=ndimage.label(left!=0,structure=S8)
    temps=[]
    for k in range(1,n+1):
        ys,xs=np.where(lab==k)
        col=int(np.bincount(left[ys,xs]).argmax())
        r0,r1=ys.min(),ys.max(); c0,c1=xs.min(),xs.max()
        pat=left[r0:r1+1,c0:c1+1].copy()
        pat[pat!=col]=0
        temps.append(dict(color=col,pat=pat,rows=(r0,r1),cols=(c0,c1),size=(pat!=0).sum()))
    return temps

def solve(g, order='size_desc'):

import numpy as np
def get_templates(g,wall):
    left=g[:,:wall];temps=[]
    for col in np.unique(left):
        if col==0:continue
        ys,xs=np.where(left==col)
        r0,r1,c0,c1=ys.min(),ys.max(),xs.min(),xs.max()
        pat=(left[r0:r1+1,c0:c1+1]==col).astype(int)*int(col)
        temps.append(dict(color=int(col),pat=pat,rows=(r0,r1),cols=(c0,c1),
                          coords=list(zip(ys,xs)),size=int((pat!=0).sum())))
    return temps
def solve(g,order='color_desc'):
    g=np.array(g).copy();H,W=g.shape
    wall=None
    for c in range(W):
        col=g[:,c]
        if col[0]!=0 and np.all(col==col[0]):wall=c;break
    temps=get_templates(g,wall)
    r0,r1=0,H-1;c0=wall+1;c1=W-1
    to=[]
    for t in temps:
        rows=[r for r in range(r0+1,r1) if g[r,c0]==t['color']]
        cols=[cc for cc in range(c0+1,c1) if g[r0,cc]==t['color']]
        if rows and cols:
            to.append((t,rows,cols))
            for (yy,xx) in t['coords']:g[yy,xx]=0
    key={'size_desc':lambda z:(-z[0]['size'],z[0]['color']),
         'size_asc':lambda z:(z[0]['size'],z[0]['color']),
         'color_desc':lambda z:(-z[0]['color'],),
         'color_asc':lambda z:(z[0]['color'],)}[order]
    to.sort(key=key)
    for t,rows,cols in to:
        pat=t['pat'];h,w=pat.shape;ch,cw=h//2,w//2
        for r in rows:
            for cc in cols:
                tr,tc=r-ch,cc-cw
                for i in range(h):
                    for j in range(w):
                        if pat[i,j]!=0:g[tr+i,tc+j]=pat[i,j]
    return g
