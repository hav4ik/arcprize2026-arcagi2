import numpy as np
from collections import deque

def components_nonzero(g):
    H,W=g.shape
    seen=np.zeros((H,W),bool); comps=[]
    for i in range(H):
        for j in range(W):
            if g[i,j]!=0 and not seen[i,j]:
                q=deque([(i,j)]); seen[i,j]=True; cells=[]
                while q:
                    r,c=q.popleft(); cells.append((r,c))
                    for dr,dc in ((1,0),(-1,0),(0,1),(0,-1)):
                        nr,nc=r+dr,c+dc
                        if 0<=nr<H and 0<=nc<W and g[nr,nc]!=0 and not seen[nr,nc]:
                            seen[nr,nc]=True; q.append((nr,nc))
                comps.append(cells)
    return comps

def find_palette(g):
    for cells in components_nonzero(g):
        rs=[r for r,c in cells]; cs=[c for r,c in cells]
        r0,r1=min(rs),max(rs); c0,c1=min(cs),max(cs)
        h=r1-r0+1; w=c1-c0+1
        if len(cells)==h*w and ((h==2 and w>=2) or (w==2 and h>=2)):
            sub=g[r0:r1+1,c0:c1+1]
            if not np.any(sub==0):
                return cells,(r0,c0,h,w),sub
    return None
