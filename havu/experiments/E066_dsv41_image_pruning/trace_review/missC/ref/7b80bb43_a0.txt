def analyze(g, bg):
    a=np.array(g)
    col = [v for v in np.unique(a) if v!=bg][0]
    H,W=a.shape
    M=(a==col).astype(int)
    def gaps_line(arr):
        # arr 1D bool; find empty runs of length exactly 3 between two filled cells
        res=[]
        n=len(arr); i=0
        while i<n:
            if arr[i]==0:
                j=i
                while j<n and arr[j]==0: j+=1
                if i>0 and j<n and arr[i-1] and arr[j] and j-i==3:
                    res.append((i,j-1))
                i=j

def solve(g):

    vals,cnts=np.unique(a,return_counts=True)
    bg=vals[np.argmax(cnts)]

    M=(a==fg)
    line=np.zeros_like(M)
    for r in range(H):
        for c in range(W):
            if M[r,c] and ((c>0 and M[r,c-1]) or (c<W-1 and M[r,c+1]) or (r>0 and M[r-1,c]) or (r<H-1 and M[r+1,c])):
                line[r,c]=True
    diag=M&~line
    lab,n=label(M,structure=np.ones((3,3)))
    sizes={k:np.sum(lab==k) for k in range(1,n+1)}
    removable=np.zeros_like(M)
    for r,c in np.argwhere(diag):
        if sizes[lab[r,c]]>1:

    # gaps length3
    def gaps_line(arr):
        res=[]; n=len(arr); i=0
        while i<n:
            if not arr[i]:
                j=i
                while j<n and not arr[j]: j+=1
                if i>0 and j<n and arr[i-1] and arr[j] and j-i==3:
                    res.append((i,j-1))
                i=j
            else: i+=1
        return res
    gaps=[]

    out=M.copy()

    for typ,idx,s,e in gaps:

import numpy as np
from scipy.ndimage import label

def solve(g, debug=False):
    a=np.array(g); H,W=a.shape
    vals,cnts=np.unique(a,return_counts=True)
    bg=vals[np.argmax(cnts)]
    fg=[v for v in vals if v!=bg][0]
    M=(a==fg)
    line=np.zeros_like(M)
    for r in range(H):
        for c in range(W):
            if M[r,c]:
                h=(c>0 and M[r,c-1]) or (c<W-1 and M[r,c+1])
                v=(r>0 and M[r-1,c]) or (r<H-1 and M[r+1,c])
                if h or v: line[r,c]=True
    diag=M&~line
    lab,n=label(M,structure=np.ones((3,3)))
    comp_line={k:False for k in range(1,n+1)}
    for r,c in np.argwhere(line): comp_line[lab[r,c]]=True
    sizes={k:np.sum(lab==k) for k in range(1,n+1)}
    removable=np.zeros_like(M)
    eligible=np.zeros_like(M)
    for r,c in np.argwhere(diag):
        if sizes[lab[r,c]]>1:
            removable[r,c]=True
            if comp_line[lab[r,c]]:
                eligible[r,c]=True
    def gaps_line(arr):
        res=[]; n=len(arr); i=0
        while i<n:
            if not arr[i]:
                j=i
                while j<n and not arr[j]: j+=1
                if i>0 and j<n and arr[i-1] and arr[j] and j-i==3:
                    res.append((i,j-1))
                i=j
            else: i+=1
        return res
    gaps=[]
    for r in range(H):
        for s,e in gaps_line(M[r]): gaps.append(('row',r,s,e))
    for c in range(W):
        for s,e in gaps_line(M[:,c]): gaps.append(('col',c,s,e))
    out=M.copy()
    elig=set(map(tuple,np.argwhere(eligible)))
    filled=[]
    for typ,idx,s,e in gaps:
        cells=[(idx,x) for x in range(s,e+1)] if typ=='row' else [(x,idx) for x in range(s,e+1)]
        near=any((gc[0]+dr,gc[1]+dc) in elig for gc in cells for dr in (-1,0,1) for dc in (-1,0,1))
        if not near: continue
        # aligned flank check
        flanks=[cells[0],cells[-1]]
        aligned=False
        for (fr,fc) in flanks:
            if typ=='row':
                if (fc>0 and line[fr,fc-1]) or (fc<W-1 and line[fr,fc+1]): aligned=True
            else:
                if (fr>0 and line[fr-1,fc]) or (fr<H-1 and line[fr+1,fc]): aligned=True
        if aligned:
            for gc in cells: out[gc]=True
            filled.append((typ,idx,s,e))
    for r,c in np.argwhere(removable): out[r,c]=False
    if debug: print('filled',filled)
    res=np.where(out,fg,bg)
    return res.astype(int)
