def solve(a):
    a=np.array(a); vals=set(a.ravel()); bg=max(vals,key=lambda v:(a==v).sum()); O=[v for v in vals if v!=bg][0]

    noise=np.zeros_like(obj)

            if obj[y,x]:
                for dy,dx in [(1,1),(1,-1)]:
                    # chain containing cell in direction; check prev/next
                    prev=(y-dy,x-dx)
                    nxt=(y+dy,x+dx)
                    has_prev = 0<=prev[0]<R and 0<=prev[1]<C and obj[prev]
                    has_next = 0<=nxt[0]<R and 0<=nxt[1]<C and obj[nxt]

    # Actually diagonal run length>=2: any cell with diagonal neighbor (both directions)
