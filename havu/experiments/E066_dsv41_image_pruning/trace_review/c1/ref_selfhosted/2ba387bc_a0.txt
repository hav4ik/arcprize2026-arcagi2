from scipy import ndimage

def parse(grid):
    g=np.array(grid)
    objs=[]
    for c in set(g.ravel())-{0}:
        mask=(g==c)
        lbl,n=ndimage.label(mask)
        for i in range(1,n+1):
            ys,xs=np.where(lbl==i)
            r,cmin=ys.min(),xs.min()
            sub=mask[r:ys.max()+1,cmin:xs.max()+1]
            solid = sub.all()
            objs.append((r,cmin,c,solid))
    objs.sort()
    return objs
