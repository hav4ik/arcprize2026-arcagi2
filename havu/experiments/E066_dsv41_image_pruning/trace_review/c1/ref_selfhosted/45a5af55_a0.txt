from itertools import groupby
import numpy as np

def solve(grid):
    rows=[tuple(r) for r in grid]
    colors=[]
    lengths=[]
    for c,g in groupby(rows):
        # each row is uniform; use color's first cell
        col=c[0]
        n=sum(1 for _ in g)
        colors.append(col)
        lengths.append(n)
    # if same color adjacent after groupby? groupby already merges identical row tuples, not color. rows are uniform so same.
    S=2*sum(lengths[:-1])+lengths[-1]
    out=np.zeros((S,S),dtype=int)
    off=0
    for col,t in zip(colors,lengths):
        out[off:S-off,off:S-off]=col
        off+=t
    return out
