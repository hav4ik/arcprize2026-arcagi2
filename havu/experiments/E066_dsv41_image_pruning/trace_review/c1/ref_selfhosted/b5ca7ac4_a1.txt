import numpy as np
from collections import Counter

def find_objs(g):
    a=np.array(g); H,W=a.shape
    objs=[]
    for r in range(H-4):
      for c in range(W-4):
        border=a[r,c]
        if border not in (2,8): continue
        # perimeter uniform border
        per=[]
        for cc in range(c,c+5): per += [a[r,cc],a[r+4,cc]]
        for rr in range(r+1,r+4): per += [a[rr,c],a[rr,c+4]]
        if not all(x==border for x in per): continue
        inner=a[r+1:r+4,c+1:c+4]
        if np.all(inner==inner[0,0]) and inner[0,0]!=border:
          objs.append((r,c,border,int(inner[0,0])))
    # there may be candidates shifted? dedup exact
    return objs

def solve(g):
    a=np.array(g); H,W=a.shape
    objs=find_objs(g)
    # get bg
    bg=Counter(a.flatten()).most_common(1)[0][0]
    out=np.full_like(a,bg)
    groups={2:[],8:[]}
    for r,c,b,inn in objs:
        groups[b].append((r,c,b,inn))
    finals={}
    # left
    for r,c,b,inn in sorted(groups[8], key=lambda x:x[1]):
        fc=0
        for rr,cc,bb,ii in groups[8]:
            if (rr,cc)==(r,c): continue
            # use final if computed
            if (rr,cc) in finals and not (rr+4<r or r+4<rr):
                fc=max(fc, finals[(rr,cc)][1]+5)
        finals[(r,c)]=(r,fc)
    # right
    for r,c,b,inn in sorted(groups[2], key=lambda x:-x[1]):
        fc=W-5
        for rr,cc,bb,ii in groups[2]:
            if (rr,cc)==(r,c): continue
            if (rr,cc) in finals and not (rr+4<r or r+4<rr):
                fc=min(fc, finals[(rr,cc)][1]-5)
        finals[(r,c)]=(r,fc)
    # paint
    for r,c,b,inn in objs:
        fr,fc=finals[(r,c)]
        # paint border and inner
        out[fr:fr+5,fc:fc+5]=b
        out[fr+1:fr+4,fc+1:fc+4]=inn
    return out,objs,finals
