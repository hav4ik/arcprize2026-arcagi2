def find_tiling(g):
    H,W=g.shape
    # rows that are uniform (all same)
    unrows=[i for i in range(H) if len(set(g[i]))==1]
    uncols=[j for j in range(W) if len(set(g[:,j]))==1]
    # separators are all uniform incl edges; interior tiles between
    return unrows,uncols

def tiles_from_uniform(g, unrows, uncols):
    rs=[(unrows[i]+1,unrows[i+1]) for i in range(len(unrows)-1) if unrows[i+1]>unrows[i]+1]

    return [[g[r0:r1,c0:c1] for c0,c1 in cs] for r0,r1 in rs]
