from scipy import ndimage

def solve(g):
    g=np.array(g)
    bg=np.bincount(g.flatten()).argmax()
    out=np.full_like(g,bg)
    comps=[]
    # label each color separately
    for c in np.unique(g):
        if c==bg: continue
        lab,n=ndimage.label(g==c, structure=np.ones((3,3)))
        for i in range(1,n+1):
            ys,xs=np.where(lab==i)
            comps.append((c,len(ys),(ys.min(),ys.max(),xs.min(),xs.max())))
    frames=[t for t in comps if t[1]>=28 and t[2][1]-t[2][0]+1>=5 and t[2][3]-t[2][2]+1>=11]
    small=[t for t in comps if t not in frames]
    for c,s,b in frames:
        y0,y1,x0,x1=b
        # draw frame pixels (component may be ring)
        mask=(g==c)
        # but need connected comp: reconstruct
        lab,n=ndimage.label(mask, structure=np.ones((3,3)))
        for i in range(1,n+1):
            ys,xs=np.where(lab==i)
            if (ys.min(),ys.max(),xs.min(),xs.max())==b:
                out[ys,xs]=c
        N=sum(1 for cc,ss,bb in small if cc==c)
        cy=(y0+y1)//2
        positions=[x0+2,x0+4,x0+6,x0+8]
        for px in positions[-N:] if N>0 else []:
            out[cy,px]=c
    return out
