import numpy as np
def shapes(g):
    g=np.array(g)
    comps=[]
    seen=np.zeros(g.shape,bool)
    for r in range(g.shape[0]):
        for c in range(g.shape[1]):
            if g[r,c]!=0 and not seen[r,c]:
                col=g[r,c]
                # flood
                stack=[(r,c)]; seen[r,c]=True; cells=[]
                while stack:
                    x,y=stack.pop(); cells.append((x,y))
                    for dx in [-1,0,1]:
                        for dy in [-1,0,1]:
                            nx,ny=x+dx,y+dy
                            if 0<=nx<g.shape[0] and 0<=ny<g.shape[1] and not seen[nx,ny] and g[nx,ny]==col:
                                seen[nx,ny]=True; stack.append((nx,ny))
                rs=[p[0] for p in cells]; cs=[p[1] for p in cells]
                comps.append((col,min(rs),max(rs),min(cs),max(cs),len(cells)))
    return sorted(comps,key=lambda t:(t[1],t[3]))
