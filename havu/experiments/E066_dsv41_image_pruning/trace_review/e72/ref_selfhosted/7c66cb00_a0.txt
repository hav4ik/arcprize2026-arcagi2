import numpy as np
from scipy import ndimage

def solve(g):
    g=np.array(g); H,W=g.shape
    bg=np.bincount(g.flatten()).argmax()
    mask=(g!=bg)
    lab,n=ndimage.label(mask, structure=np.ones((3,3)))
    comps=[]
    for i in range(1,n+1):
        ys,xs=np.where(lab==i)
        comps.append(dict(id=i,ys=ys,xs=xs,bbox=(ys.min(),ys.max(),xs.min(),xs.max()),
                          size=len(ys),colors=set(g[lab==i].tolist())))
    out=g.copy()
    rects=[c for c in comps if (c['bbox'][3]-c['bbox'][2]+1)==W]
    stamps=[c for c in comps if c not in rects]
    # determine fill/corner for rects
    for r in rects:
        vals=g[lab==r['id']]
        bc=np.bincount(vals)
        fill=bc.argmax(); corner=[v for v in r['colors'] if v!=fill]
        r['fill']=fill; r['corner']=corner[0]
    # erase stamps
    for s in stamps:
        for y,x in zip(s['ys'],s['xs']):
            out[y,x]=bg
    # paint
    for s in stamps:
        ys,xs=s['ys'].astype(int),s['xs'].astype(int)
        scolors=np.array([g[y,x] for y,x in zip(ys,xs)])
        for r in rects:
            F=r['fill']; M=r['corner']
            sel=(scolors==F)
            if not sel.any(): continue
            sy=ys[sel]; sx=xs[sel]
            rmin,cmin=sy.min(),sx.min()
            h=sy.max()-rmin+1
            bottom=r['bbox'][1]
            top=bottom-h+1
            for y,x in zip(sy,sx):
                ny=top+(y-rmin); nx=x
                out[ny,nx]=M
    return out
