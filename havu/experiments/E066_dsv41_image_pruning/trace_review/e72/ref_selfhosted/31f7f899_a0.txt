from collections import deque
import numpy as np

def solve(grid):
    g=np.array(grid)
    bg=8
    # line row = row with most 6s
    line_row=int(np.argmax((g==6).sum(axis=1)))
    H,W=g.shape
    # connected components of cells not background and not line color (6)
    seen=np.zeros_like(g,dtype=bool)
    comps=[]
    for r in range(H):
        for c in range(W):
            if seen[r,c] or g[r,c] in (bg,6): continue
            col=int(g[r,c])
            # BFS same color
            q=[(r,c)]; seen[r,c]=True; pts=[]
            while q:
                x,y=q.pop(); pts.append((x,y))
                for dx,dy in [(1,0),(-1,0),(0,1),(0,-1)]:
                    nx,ny=x+dx,y+dy
                    if 0<=nx<H and 0<=ny<W and not seen[nx,ny] and g[nx,ny]==col:
                        seen[nx,ny]=True; q.append((nx,ny))
            rs=[p[0] for p in pts]; cs=[p[1] for p in pts]
            comps.append({'color':col,'r0':min(rs),'r1':max(rs),'c0':min(cs),'c1':max(cs),'pts':pts})
    bars=[o for o in comps if (o['r1']-o['r0']+1)>1 or (o['r0']<=line_row<=o['r1'] and o['c0']==o['c1'])]
    # Keep only components crossing/on line row and vertical-ish? In examples all bars cross line row
    bars=[o for o in bars if o['r0']<=line_row<=o['r1']]
    bars.sort(key=lambda o:(o['c0'],o['c1']))
    # spans sorted by height
    spans=sorted([(o['r1']-o['r0']+1,o['r0'],o['r1']) for o in bars], key=lambda t:(t[0],t[1]))
    out=g.copy()
    # erase bar cells
    for o in bars:
        for x,y in o['pts']:
            out[x,y]=bg
    # draw bars with assigned spans
    for o,(h,r0,r1) in zip(bars,spans):
        out[r0:r1+1, o['c0']:o['c1']+1]=o['color']
    return out
