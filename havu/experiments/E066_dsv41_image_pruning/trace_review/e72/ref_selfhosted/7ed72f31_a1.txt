from scipy.ndimage import label

def objects(I,bg):
    mask=(I!=bg)&(I!=2)
    lab,n=label(mask,structure=np.ones((3,3)))
    objs=[]
    for k in range(1,n+1):
        cells=[tuple(x) for x in np.argwhere(lab==k)]
        objs.append(cells)
    return objs

def redcells(I):
    return [tuple(x) for x in np.argwhere(I==2)]

def refl_point(cells,rc,cc):
    return [(2*rc-r,2*cc-c) for r,c in cells]
def refl_h(cells,r0):  # mirror across horizontal line row r0 (line at r0+fraction?) use axis r0 (integer cell) -> r'=2r0-r
    return [(2*r0-r,c) for r,c in cells]
def refl_v(cells,c0):
    return [(r,2*c0-c) for r,c in cells]
def refl_diag_main(cells,rc,cc):  # across r-c = rc-cc : (r,c)->(c + (rc-cc), r - (rc-cc))
    d=rc-cc
    return [(c+d, r-d) for r,c in cells]
def refl_diag_anti(cells,rc,cc):
    s=rc+cc
    return [(s-c, s-r) for r,c in cells]
