    def comps(m):

            ys,xs=np.where(lab==k)
            res.append((ys,xs))
        return res
    # components of non-bg

    for c in np.unique(g):

    bars=[(c,ys,xs) for c,ys,xs in allc if len(ys)==5 and ys.max()==ys.min()]
