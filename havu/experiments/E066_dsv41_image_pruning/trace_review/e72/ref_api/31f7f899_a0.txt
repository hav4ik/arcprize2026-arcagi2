def bars(a):
    H,W=a.shape
    mask=~np.isin(a,[8,6])
    groups=[]
    c=0
    while c<W:
        if mask[:,c].any():
            color=int(np.bincount(a[mask[:,c],c]).argmax())
            rows=np.where(mask[:,c])[0]
            c2=c
            cols=[c]
            while c2+1<W and mask[:,c2+1].any() and int(np.bincount(a[mask[:,c2+1],c2+1]).argmax())==color:

            allrows=np.where(mask[:,cols].any(axis=1))[0]
            groups.append((cols,color,len(allrows)))
            c=c2+1

            c+=1
    return groups
