import numpy as np
from collections import Counter, deque

def comps(g, bg):
    H,W=g.shape
    seen=np.zeros_like(g,bool)
    out=[]
    for i in range(H):
        for j in range(W):
            if g[i,j]!=bg and not seen[i,j]:
                q=deque([(i,j)]); seen[i,j]=True; cells=[]
                while q:
                    r,c=q.popleft(); cells.append((r,c))
                    for dr,dc in((1,0),(-1,0),(0,1),(0,-1)):
                        nr,nc=r+dr,c+dc
                        if 0<=nr<H and 0<=nc<W and g[nr,nc]!=bg and not seen[nr,nc]:
                            seen[nr,nc]=True; q.append((nr,nc))
                out.append(cells)
    return out
