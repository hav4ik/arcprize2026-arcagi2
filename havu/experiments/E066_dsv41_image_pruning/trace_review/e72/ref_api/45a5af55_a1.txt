def get_layers(a):
    a=np.array(a)
    layers=[]
    for r in a:
        c=int(r[0])
        if layers and layers[-1][0]==c: layers[-1][1]+=1
        else: layers.append([c,1])
    return layers

def solve(a):
    layers=get_layers(a)
    S=sum(l for c,l in layers); L=layers[-1][1]
    N=2*S-L
    out=np.zeros((N,N),dtype=int)
    # cumulative starts
    starts=[]; x=0
    for c,l in layers:
        starts.append(x); x+=l
    for r in range(N):
        for col in range(N):
            d=min(r,col,N-1-r,N-1-col)
            # find layer
            for (c,l),s in zip(layers,starts):
                if s<=d<s+l:
                    out[r,col]=c; break
            else:
                out[r,col]=layers[-1][0]
    return out
