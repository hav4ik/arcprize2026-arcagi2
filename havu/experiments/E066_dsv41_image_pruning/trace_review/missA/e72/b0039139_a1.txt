def segs(a):

def parse_parts(a):
    a=np.array(a); H,W=a.shape
    sep_rows=[r for r in range(H) if np.all(a[r]==1)]
    sep_cols=[c for c in range(W) if np.all(a[:,c]==1)]
    if sep_rows:
        cuts=[]; prev=-1
        for r in sep_rows:
            cuts.append(a[prev+1:r]); prev=r
        cuts.append(a[prev+1:])
        ori=0
    else:
        cuts=[]; prev=-1
