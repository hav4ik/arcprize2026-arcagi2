from collections import deque, defaultdict

def solve_grid(a, debug=False):
    a=np.array(a)
    vals,counts=np.unique(a,return_counts=True)
    bg=vals[np.argmax(counts)]
    # sep = most frequent non-bg
    sep=max([v for v in vals if v!=bg], key=lambda v: counts[list(vals).index(v)])
    mask=(a!=sep)
    lab,n=label(mask) # 4-connectivity default
    out=a.copy()
    H,W=a.shape
    for cid in range(1,n+1):
        comp=(lab==cid)
        # seeds
        seed_cells=[]
        for r,c in np.argwhere(comp):
            v=a[r,c]
            if v!=bg and v!=sep:
                seed_cells.append((r,c,int(v)))
        if not seed_cells:
            continue
        # 8-neighbor peeling depth within comp
        depth=np.full((H,W),-1,int); q=deque()
        for r,c in np.argwhere(comp):
            border=False
            for dr in (-1,0,1):
                for dc in (-1,0,1):
                    if dr==0 and dc==0: continue
                    nr,nc=r+dr,c+dc
                    if not(0<=nr<H and 0<=nc<W) or not comp[nr,nc]:
                        border=True
            if border:
                depth[r,c]=0; q.append((r,c))
        while q:
            r,c=q.popleft()
            for dr in (-1,0,1):
                for dc in (-1,0,1):
                    if dr==0 and dc==0: continue
                    nr,nc=r+dr,c+dc
                    if 0<=nr<H and 0<=nc<W and comp[nr,nc] and depth[nr,nc]<0:

        byd={}

        for r,c in np.argwhere(comp):

        if debug:
