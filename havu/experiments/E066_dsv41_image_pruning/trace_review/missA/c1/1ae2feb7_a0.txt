import numpy as np

def find_div(a):
    H,W=a.shape
    best=None
    for c in range(W):
        col=a[:,c]
        vals,cnts=np.unique(col[col!=0],return_counts=True)
        for v,n in zip(vals,cnts):
            # count rows with value v in col
            if n>=max(1,H//2):
                if best is None or n>best[2]:
                    best=(c,int(v),int(n))
    return best

def solve(g):
    a=np.array(g); H,W=a.shape
    c,dc,_=find_div(a)
    # counts nonzero on each side
    left_count=np.count_nonzero(a[:,:c])
    right_count=np.count_nonzero(a[:,c+1:])
    source='left' if left_count>=right_count else 'right'
    b=a.copy()
    if source=='left':
        # runs from farthest (col0) to nearest (col c-1)
        for r in range(H):
            seg=a[r,:c]
            runs=[]
            x=0; L=c
            while x<L:
                if seg[x]==0: x+=1; continue
                col=seg[x]; y=x
                while y<L and seg[y]==col: y+=1
                runs.append((col,y-x))
                x=y
            # far-to-near already left->right
            right_len=W-c-1
            for col,L in runs:
                for off in range(0,right_len,L):
                    b[r,c+1+off]=col
    else:
        right_len=W-c-1
        for r in range(H):
            seg=a[r,c+1:]
            runs=[]
            x=0
            while x<len(seg):
                if seg[x]==0: x+=1; continue
                col=seg[x]; y=x
                while y<len(seg) and seg[y]==col: y+=1
                runs.append((col,y-x,x)) # include start distance
                x=y
            # process far-tonear: rightmost to leftmost
            for col,L,st in sorted(runs, key=lambda t:t[2], reverse=True):
                for off in range(0,c,L):
                    b[r,c-1-off]=col
    return b
