from scipy.ndimage import label

def solve(a):
    a=np.array(a)
    H,W=a.shape
    # line row(s) color6; assume one row with count
    # components of non 8 and non6

    lab,n=label(mask, structure=np.ones((3,3)))
    comps=[]
    for k in range(1,n+1):
        ys,xs=np.where(lab==k)
        if len(ys)==0: continue
        col=a[ys,xs][0]
        comps.append({'color':int(col),'xmin':int(xs.min()),'xmax':int(xs.max()),'ymmin':int(ys.min()),'ymmax':int(ys.max()),'h':int(ys.max()-ys.min()+1)})
    # horizontal row: row where color6 occurs most
    row=int(np.argmax([np.sum(a[r]==6) for r in range(H)]))
    comps=sorted(comps,key=lambda c:(c['xmin'],c['xmax']))
    hs=sorted([c['h'] for c in comps])
    out=np.full((H,W),8,d=int)
    # draw horizontal line full row
    out[row,:]=6
    for c,h in zip(comps,hs):
        start=row-h//2
        end=start+h-1
        out[start:end+1,c['xmin']:c['xmax']+1]=c['color']
    return out
