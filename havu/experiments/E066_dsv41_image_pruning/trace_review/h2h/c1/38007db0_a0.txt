def cells(a, rows, cols):
    # rows/cols separator positions include borders
    R=[(rows[i]+1,rows[i+1]) for i in range(len(rows)-1)]
    C=[(cols[i]+1,cols[i+1]) for i in range(len(cols)-1)]
    return [[a[r[0]:r[1], c[0]:c[1]] for c in C] for r in R]

def sep(a):
    rows=[i for i in range(a.shape[0]) if len(set(a[i]))==1]
    cols=[j for j in range(a.shape[1]) if len(set(a[:,j]))==1]
    return rows,cols
