from scipy import ndimage

def solve(g):
    g=np.array(g)
    H,W=g.shape
    line_row=int(np.argmax((g==6).sum(axis=1)))
    mask=(g!=8)&(g!=6)
    lbl,n=ndimage.label(mask, structure=np.ones((3,3)))
    bars=[]
    for i in range(1,n+1):
        ys,xs=np.where(lbl==i)
        if len(xs)==0: continue
        color=int(g[ys[0],xs[0]])
        bars.append((xs.min(),xs.max(),ys.min(),ys.max(),len(ys),color))
    bars.sort(key=lambda t:t[0])
    heights=sorted([b[5] for b in bars])
    out=np.full((H,W),8,dtype=int)
    out[line_row,:]=6
    for b,h in zip(bars,heights):
        x0,x1,_,_,_,color=b
        top=line_row-h//2
        out[top:top+h, x0:x1+1]=color
    return out
