import numpy as np

def solve(a):
    a=np.array(a)
    # runs of horizontal bands (each row uniform)
    runs=[]
    prev=None
    for row in a:
        c=int(row[0])
        if c==prev:
            runs[-1][1]+=1
        else:
            runs.append([c,1]); prev=c
    # build from innermost (last run) outward
    c,t = runs[-1]
    g=np.full((t,t),c,dtype=int)
    for c,t in reversed(runs[:-1]):
        n=g.shape[0]+2*t
        ng=np.full((n,n),c,dtype=int)
        ng[t:t+g.shape[0], t:t+g.shape[1]]=g
        g=ng
    return g
