from scipy import ndimage
def comps(g):
    bg=np.bincount(g.ravel()).argmax()
    res=[]
    for c in np.unique(g):
        if c==bg: continue
        lab,n=ndimage.label(g==c, structure=np.ones((3,3)))

            res.append((c,ys.min(),ys.max(),xs.min(),xs.max(),len(ys)))
