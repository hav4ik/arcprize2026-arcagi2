from scipy import ndimage

def solve(grid):
    a=np.array(grid).copy()
    H,W=a.shape
    bg=np.bincount(a.ravel()).argmax()
    # detect canvas rows: rows where a[r,0]==a[r,W-1] and that equals border; canvas fill is interior
    # canvases are full-width rows with a[r,0]==a[r,W-1] and a[r,1]==... and border differs from fill
    cand=np.where((a[:,0]==a[:,W-1]) & (a[:,0]!=bg))[0]
    # group consecutive
    groups=[]
    for r in cand:
        if groups and r==groups[-1][-1]+1: groups[-1].append(r)
        else: groups.append([r])
    canv=[]
    for g in groups:
        r0,r1=g[0],g[-1]
        B=a[r0,0]; F=a[r0,1]
        # verify border col only ends? and interior consistent
        canv.append((r0,r1,B,F))
    canvas_rows=set()
    for r0,r1,B,F in canv: canvas_rows.update(range(r0,r1+1))
    toprows=[r for r in range(H) if r not in canvas_rows]
    # top region may include bottom? but patterns only above canv; restrict to rows before min canvas
    if canv: toprows=[r for r in toprows if r<canv[0][0]]
    for r0,r1,B,F in canv:
        mask=np.zeros((H,W),bool)
        for r in toprows:
            mask[r,:]=(a[r,:]==F)
        lab,n=ndimage.label(mask)
        comps=[]
        for k in range(1,n+1):
            ys,xs=np.where(lab==k)
            comps.append((ys,xs))
        for ys,xs in comps:
            yshift=r1-ys.max()
            a[ys+yshift, xs]=B
    return a
