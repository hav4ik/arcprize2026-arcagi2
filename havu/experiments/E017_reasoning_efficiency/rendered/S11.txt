import numpy as np

def split_blocks(I):
    # split by rows/cols that are entirely 1
    I=np.array(I)
    # find all-1 rows
    row1=[i for i in range(I.shape[0]) if np.all(I[i]==1)]
    col1=[j for j in range(I.shape[1]) if np.all(I[:,j]==1)]
    # split rows into segments between row1
    rowsegs=[]; prev=-1
    for r in row1+[I.shape[0]]:
        if r-prev>1: rowsegs.append((prev+1,r))
        prev=r
    colsegs=[]; prev=-1
    for c in col1+[I.shape[1]]:
        if c-prev>1: colsegs.append((prev+1,c))
        prev=c
    return rowsegs,colsegs

import numpy as np

def split_blocks(I):
    I=np.array(I)
    row1=[i for i in range(I.shape[0]) if np.all(I[i]==1)]
    col1=[j for j in range(I.shape[1]) if np.all(I[:,j]==1)]
    rowsegs=[]; prev=-1
    for r in row1+[I.shape[0]]:
        if r-prev>1: rowsegs.append((prev+1,r))
        prev=r
    colsegs=[]; prev=-1
    for c in col1+[I.shape[1]]:
        if c-prev>1: colsegs.append((prev+1,c))
        prev=c
    return rowsegs,colsegs

def bbox_crop(g, vals):
    ys,xs=np.where(np.isin(g, vals))
    return g[ys.min():ys.max()+1, xs.min():xs.max()+1]

def solve(I):
    I=np.array(I)
    rs,cs=split_blocks(I)
    ps=[]
    for (r0,r1) in rs:
        for (c0,c1) in cs:
            ps.append(I[r0:r1,c0:c1])
    shape=bbox_crop(ps[0],[4])
    pat=ps[1]
    A=int(ps[2][0,0]); B=int(ps[3][0,0])
    vertical = len(rs)>1 and len(cs)==1
    if vertical:
        # count distinct rows containing 3
        rows3=sorted(set(np.where(np.any(pat==3,axis=1))[0].tolist()))
        N=len(rows3)
        # recolor shape
        tile=np.where(shape==4, A, B)
        sep=np.full((1,tile.shape[1]),B)
        blocks=[tile]*N
        out=np.vstack([x for i,b in enumerate(blocks) for x in ([b,sep] if i<N-1 else [b])])
    else:
        cols3=sorted(set(np.where(np.any(pat==3,axis=0))[0].tolist()))
        N=len(cols3)
        tile=np.where(shape==4, A, B)
        sep=np.full((tile.shape[0],1),B)
        blocks=[tile]*N
        out=np.hstack([x for i,b in enumerate(blocks) for x in ([b,sep] if i<N-1 else [b])])
    return out
