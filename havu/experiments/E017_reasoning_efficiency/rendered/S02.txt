import numpy as np

def get_bars(inp):
    H,W=inp.shape
    bars=[]
    for r in range(H):
        c=0
        while c<W:
            if inp[r,c]==2:
                s=c
                while c<W and inp[r,c]==2:
                    c+=1
                if c-s>=2:
                    bars.append((r,s,c-1))  # row, left, right
            else:
                c+=1
    return bars

    marker_bottom={}

    # function: nearest bar below given row with top span covering x
