import numpy as np

def line_fill(cells, L, s):
    # cells: list of (idx,color); s=+1 extend toward +, s=-1 toward -
    if len(cells)<2: return {}
    from collections import Counter
    cnt=Counter(c for _,c in cells)
    bases=[c for c,k in cnt.items() if k>=2]
    if len(bases)!=1: return {}
    base=bases[0]
    others=[(i,c) for i,c in cells if c!=base]
    if len(set(c for _,c in others))>1: return {}
    bidx=sorted(i for i,c in cells if c==base)
    d=bidx[1]-bidx[0]
    for a,b in zip(bidx,bidx[1:]):
        if b-a!=d: return {}
    res={i:base for i in bidx}
    if len(others)==1:
        ti,tc=others[0]
        extreme = max(bidx) if s>0 else min(bidx)
        beyond = (ti-extreme)*s>0
        aligned = (ti-bidx[0])%d==0
        if beyond and aligned:
            x=ti
            while 0<=x<L:
                res[x]=tc
                x-=s*d
            return res
    # no usable tip: extend
    extreme = max(bidx) if s>0 else min(bidx)
    x=extreme+s*d
    while 0<=x<L:
        res[x]=base
        x+=s*d
    return res

def solve(grid):
    a=np.array(grid)
    H,W=a.shape
    out=np.zeros((H,W),dtype=int)
    # rows
    for r in range(H):
        cells=[(c,int(a[r,c])) for c in range(W) if a[r,c]!=0]
        for i,c in line_fill(cells,W,+1).items():
            out[r,i]=c
    # cols
    for c in range(W):
        cells=[(r,int(a[r,c])) for r in range(H) if a[r,c]!=0]
        for i,col in line_fill(cells,H,-1).items():
            out[i,c]=col
    # keep? maybe also original cells that weren't filled? check
    return out

from collections import Counter

def line_fill(cells, L, default_s):
    if len(cells)<2: return {}
    cnt=Counter(c for _,c in cells)
    bases=[c for c,k in cnt.items() if k>=2]
    if len(bases)!=1: return {}
    base=bases[0]
    others=[(i,c) for i,c in cells if c!=base]
    if len(set(c for _,c in others))>1: return {}
    bidx=sorted(i for i,c in cells if c==base)
    d=bidx[1]-bidx[0]
    for a,b in zip(bidx,bidx[1:]):
        if b-a!=d: return {}
    res={i:base for i in bidx}
    if len(others)==1:
        ti,tc=others[0]
        if ti>max(bidx): s=+1
        elif ti<min(bidx): s=-1
        else: s=default_s
        aligned=(ti-bidx[0])%d==0
        if aligned:
            x=ti
            while 0<=x<L:
                res[x]=tc
                x-=s*d
            return res
        # not aligned: base extends toward tip
        extreme=max(bidx) if s>0 else min(bidx)
        x=extreme+s*d
        while 0<=x<L:
            res[x]=base
            x+=s*d
        return res
    # no tip
    s=default_s
    extreme=max(bidx) if s>0 else min(bidx)
    x=extreme+s*d
    while 0<=x<L:
        res[x]=base
        x+=s*d
    return res

def solve(grid):

        for i,col in line_fill(cells,H,-1).items(): out[i,c]=col
