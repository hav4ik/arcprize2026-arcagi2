def analyze(g):
    bg=g[0,0]
    mask=(g!=bg); lab,n=ndimage.label(mask,structure=np.ones((3,3)))
    objs=[]
    for k in range(1,n+1):
        ys,xs=np.where(lab==k)
        r0,r1,c0,c1=ys.min(),ys.max(),xs.min(),xs.max()
        crop=g[r0:r1+1,c0:c1+1].copy()
        objs.append(dict(r0=r0,c0=c0,h=r1-r0+1,w=c1-c0+1,area=len(ys),crop=crop))
    return bg,objs

def core_centroid(crop,bg):
    h,w=crop.shape
    ys,xs=[],[]
    for r in range(h):
        for c in range(w):
            v=crop[r,c]
            if v==1 or v==bg: continue
            left=any(crop[r,cc]==1 for cc in range(c))
            right=any(crop[r,cc]==1 for cc in range(c+1,w))
            up=any(crop[rr,c]==1 for rr in range(r))
            down=any(crop[rr,c]==1 for rr in range(r+1,h))
            if (left and right) or (up and down):
                ys.append(r); xs.append(c)
    return (np.mean(ys),np.mean(xs)) if ys else None
