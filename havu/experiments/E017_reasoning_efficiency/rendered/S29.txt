def comps(grid):
    g=np.array(grid)
    res=[]
    for c in sorted(set(g.flatten())-{0}):
        mask=(g==c)

import numpy as np
def chevron_rule(inp, rotate=False, keepcolor=True):
    g=np.array(inp); H,W=g.shape
    if rotate: g=np.rot90(g,2)
    res=np.zeros((H,W),int)
    cr,cc=(H-1)/2,(W-1)/2
    for r,c in zip(*np.where((g!=0)&(g!=8))):
        col=int(g[r,c])
        # direction away from center
        dr = 1 if r>cr else (-1 if r<cr else 0)
        dc = 1 if c>cc else (-1 if c<cc else 0)
        # chevron pointing outward: if horizontal dominant?
        if abs(r-cr)>=abs(c-cc):
            # vertical direction
            base=[(r+dr,c),(r,c-1),(r,c+1)] if dr!=0 else [(r,c-1),(r+1,c),(r-1,c)]

        for rr,c2 in base:
            if 0<=rr<H and 0<=c2<W: res[rr,c2]=col
    return res
