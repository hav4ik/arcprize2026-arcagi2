def solve(inp):
    inp=np.array(inp)
    H,W=inp.shape
    grid=inp.copy()
    markers=[c for c in range(W) if inp[0,c]==6 and inp[1,c]==6]
    visited=set()
    stack=[(1,c) for c in markers]
    while stack:
        r,c=stack.pop()
        if (r,c) in visited: continue
        visited.add((r,c))
        if r+1>=H: continue
        if inp[r+1,c]!=2:
            grid[r+1,c]=6
            stack.append((r+1,c))
        else:
            # run of 2 in row r+1 containing c
            l=c
            while l-1>=0 and inp[r+1,l-1]==2: l-=1
            rr=c
            while rr+1<W and inp[r+1,rr+1]==2: rr+=1
            a=max(0,l-1); b=min(W-1,rr+1)
            for cc in range(a,b+1):
                grid[r,cc]=6
            for cc in (l-1,rr+1):
                if 0<=cc<W:
                    stack.append((r,cc))
    # keep original 2s and 6s; result
    return grid
