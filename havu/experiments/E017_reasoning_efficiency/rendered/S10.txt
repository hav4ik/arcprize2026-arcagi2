def spiral(R,C,fg,bg):
    g=-np.ones((R,C),int)
    top,bot,left,right=0,R-1,0,C-1
    lap=0
    while top<=bot and left<=right:
        col = fg if lap%2==0 else bg
        for c in range(left,right+1): g[top,c]=col
        for r in range(top,bot+1): g[r,right]=col
        if top<bot:
            for c in range(right,left-1,-1): g[bot,c]=col
        if left<right:
            for r in range(bot,top,-1): g[r,left]=col
        top+=1;bot-=1;left+=1;right-=1;lap+=1
    return g

import itertools
def search_subsample(a,b):
    H,W=a.shape;h,w=b.shape
    best=[]
    for s in range(1,8):
        for oy in range(s):
            for ox in range(s):
                ys=[oy+s*i for i in range(h)]
                xs=[ox+s*j for j in range(w)]
                if ys[-1]<H and xs[-1]<W:
                    sub=a[np.ix_(ys,xs)]
                    m=(sub==b).mean()
                    best.append((m,s,oy,ox,'id'))
                    # flips
                    for ft in ['fliplr','flipud','T']:
                        sb=sub
                        if ft=='fliplr': sb=sub[:, ::-1]
                        if ft=='flipud': sb=sub[::-1]
                        if ft=='T': sb=sub.T
                        if sb.shape==b.shape:
                            best.append(((sb==b).mean(),s,oy,ox,ft))
    best.sort(reverse=True)
    return best[:5]

def spiral_order(R,C):
    g=-np.ones((R,C),int)
    r=c=0; d=0
    dirs=[(0,1),(1,0),(0,-1),(-1,0)]
    idx=0
    for _ in range(R*C):
        g[r,c]=idx; idx+=1
        nr,nc=r+dirs[d][0], c+dirs[d][1]
        if 0<=nr<R and 0<=nc<C and g[nr,nc]==-1:
            r,c=nr,nc
        else:
            d=(d+1)%4
            r,c=r+dirs[d][0], c+dirs[d][1]

def spiral_order(R,C):
    g=-np.ones((R,C),int)
    r=c=0; d=0
    dirs=[(0,1),(1,0),(0,-1),(-1,0)]
    idx=0
    for _ in range(R*C):
        g[r,c]=idx; idx+=1
        nr,nc=r+dirs[d][0], c+dirs[d][1]

        else:
            d=(d+1)%4
            r,c=r+dirs[d][0], c+dirs[d][1]
    return g

def rot45(g, bg):
    R,C=g.shape

def spiral_order(R,C,start=0,ccw=False):
    g=-np.ones((R,C),int)
    r=c=0; d=0
    dirs=[(0,1),(1,0),(0,-1),(-1,0)]
    idx=0
    for _ in range(R*C):
        g[r,c]=idx; idx+=1
        nr,nc=r+dirs[d][0], c+dirs[d][1]
        if 0<=nr<R and 0<=nc<C and g[nr,nc]==-1:
            r,c=nr,nc
        else:
            d=(d+1)%4
            r,c=r+dirs[d][0], c+dirs[d][1]
    return g

from scipy import ndimage
def maxruns(m):
    best=0
    for row in m:
        s=''.join('#' if v else '.' for v in row)
        best=max(best,len([x for x in s.split('.') if x]))
    return best
def solve(a):
    vals,counts=np.unique(a,return_counts=True); bg=vals[np.argmax(counts)]; fg=[v for v in vals if v!=bg][0]
    m=(a==fg)
    H=2*maxruns(m.T)-1; W=2*maxruns(m)-1
    ys,xs=np.where(m); oy0,oy1,ox0,ox1=ys.min(),ys.max(),xs.min(),xs.max()
    sr=(H-1)/(oy1-oy0) if oy1>oy0 else 0
    sc=(W-1)/(ox1-ox0) if ox1>ox0 else 0
    def mr(r): return int(round((r-oy0)*sr))
    def mc(c): return int(round((c-ox0)*sc))
    out=np.full((H,W),bg,int)
    lab,n=ndimage.label(m)
    for k in range(1,n+1):
        comp=(lab==k)
        cy,cx=np.where(comp)
        r0,r1,c0,c1=cy.min(),cy.max(),cx.min(),cx.max()
        holes=ndimage.binary_fill_holes(comp).sum()-comp.sum()
        if holes>0 and (r1>r0 and c1>c0):
            R0,R1,C0,C1=mr(r0),mr(r1),mc(c0),mc(c1)
            R0=max(0,min(H-1,R0)); R1=max(0,min(H-1,R1)); C0=max(0,min(W-1,C0)); C1=max(0,min(W-1,C1))
            if R1==R0: R1=min(H-1,R0+1)
            if C1==C0: C1=min(W-1,C0+1)
            out[R0,C0:C1+1]=fg; out[R1,C0:C1+1]=fg
            out[R0:R1+1,C0]=fg; out[R0:R1+1,C1]=fg

            rr=max(0,min(H-1,rr)); cc=max(0,min(W-1,cc))
            out[rr,cc]=fg
