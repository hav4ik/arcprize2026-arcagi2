def try_align(A,B):
    # find offset where B (smaller maybe) matches A
    ha,wa=A.shape; hb,wb=B.shape
    best=[]
    for dr in range(-ha,ha+1):
        for dc in range(-wa,wa+1):
            cnt=0; tot=0
            for i in range(hb):
                for j in range(wb):
                    ai=i-dr; aj=j-dc
                    if 0<=ai<ha and 0<=aj<wa:
                        tot+=1
                        if A[ai,aj]==B[i,j]: cnt+=1
            if tot>0:
                best.append((cnt/tot,cnt,tot,dr,dc))
    best.sort(reverse=True)
    return best[:5]
