import numpy as np
from collections import defaultdict

def pos(gi, colors=None):
    d=defaultdict(list)
    for r in range(gi.shape[0]):
        for c in range(gi.shape[1]):
            v=gi[r,c]
            if v!=0:
                d[int(v)].append((r,c))
    return d

def rule_diag(gi):

    occ=(gi!=0)
