WIRE={8,5}
def solve(inp):
    a=np.array(inp)
    H,Wd=a.shape
    def iswire(r,c):
        return 0<=r<H and 0<=c<Wd and a[r,c] in WIRE
    def istok(r,c):
        return 0<=r<H and 0<=c<Wd and a[r,c] not in (0,5,8,9)
    # find squares
    from scipy import ndimage
    squares=[]  # center
    tokcolors=set(a.flatten().tolist())-{0,5,8,9}
    for col in tokcolors:
        m=(a==col)
        lab,n=ndimage.label(m)
        for k in range(1,n+1):
            ys,xs=np.where(lab==k)
            if len(ys)==9:
                r0,r1,c0,c1=ys.min(),ys.max(),xs.min(),xs.max()
                if r1-r0==2 and c1-c0==2:
                    squares.append((col,(r0+1,c0+1)))
    # detect X-5 and plus-4 patterns
    Xs=[]; Ps=[]
    for r in range(H):
        for c in range(Wd):
            v=a[r,c]
            if v in (0,5,8,9): continue
            # X pattern centered at (r,c)
            if (a[r-1,c-1]==v and a[r-1,c+1]==v and a[r+1,c-1]==v and a[r+1,c+1]==v
                and a[r-1,c]!=v and a[r+1,c]!=v and a[r,c-1]!=v and a[r,c+1]!=v
                and 1<=r<H-1 and 1<=c<Wd-1):

            # plus pattern centered at (r,c) empty
            if a[r,c]==0 and 1<=r<H-1 and 1<=c<Wd-1:

    # note: for plus, need all four equal and nonzero token
    tokens=[]  # (color,center,type)
    for col,ctr in squares: tokens.append((col,ctr,'sq'))
    for col,ctr in Xs: tokens.append((col,ctr,'X'))
    for col,ctr in Ps: tokens.append((col,ctr,'P'))
    # attached wire direction
    def attach(ctr):
        r,c=ctr
        for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
            if iswire(r+2*dr,c+2*dc) and not istok(r+2*dr,c+2*dc):
