def get(i):
    ex=train[i]; g=np.array(ex['input']); o=np.array(ex['output'])
    bg=g[0,0]
    sepcols=[c for c in range(g.shape[1]) if len(set(g[:,c]))==1 and g[0,c]!=bg]
    return [(b!=bg).astype(int) for b in blocks(g,sepcols)], (o!=bg).astype(int)

def shift(m,dy,dx):
    H,W=m.shape
    sh=np.zeros_like(m)
    sh[max(0,dy):min(H,H+dy), max(0,dx):min(W,W+dx)] = m[max(0,-dy):min(H,H-dy), max(0,-dx):min(W,W-dx)]
    return sh

import numpy as np
def get(i):
    ex=train[i]; g=np.array(ex['input']); o=np.array(ex['output'])
    bg=g[0,0]
    sepcols=[c for c in range(g.shape[1]) if len(set(g[:,c]))==1 and g[0,c]!=bg]
    return [(b!=bg).astype(int) for b in blocks(g,sepcols)], (o!=bg).astype(int)
def shift(m,dy,dx):
    H,W=m.shape; sh=np.zeros_like(m)
    sh[max(0,dy):min(H,H+dy), max(0,dx):min(W,W+dx)] = m[max(0,-dy):min(H,H-dy), max(0,-dx):min(W,W-dx)]
    return sh
def find_shifts(p, om):
    H,W=om.shape
    res=[]
    for dy in range(-H+1,H):
        for dx in range(-W+1,W):
            if np.all(om <= shift(p,dy,dx)): res.append((dy,dx))
    return res
