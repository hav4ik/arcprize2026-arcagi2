def periods(g):
    g=np.array(g); h,w=g.shape
    def minp(lines):
        n=len(lines)

            if all(np.array_equal(lines[i],lines[i%p]) for i in range(n)):

        return n

import numpy as np

def nz(g): return int(np.count_nonzero(g))

def solve(grid):
    g=np.array(grid); h,w=g.shape; hh,ww=h//2,w//2
    A=g[:hh,:ww]; B=g[:hh,ww:]; C=g[hh:,:ww]; D=g[hh:,ww:]
    H=nz(A); W=nz(B)
    nc=np.argwhere(D!=0)
    if len(nc)==0:
        # 1x1 identity blocks -> output = C tiled
        big=C.copy(); dh=dw=1
    else:
        r0,c0=nc.min(0); r1,c1=nc.max(0)
        blk=D[r0:r1+1,c0:c1+1]
        dh,dw=blk.shape
        big=np.zeros((C.shape[0]*dh, C.shape[1]*dw),dtype=int)
        for I in range(C.shape[0]):
            for J in range(C.shape[1]):
                ob=np.where(blk!=0, blk, C[I,J])
                big[I*dh:(I+1)*dh, J*dw:(J+1)*dw]=ob
    # tile big to cover HxW then crop
    reps_r=int(np.ceil(H/big.shape[0])); reps_c=int(np.ceil(W/big.shape[1]))
    tiled=np.tile(big,(max(reps_r,1),max(reps_c,1)))
    return tiled[:H,:W]
