def components_mask(mask):
    H,W=mask.shape
    lab=np.zeros((H,W),int); cur=0; comps=[]

                cur+=1; stack=[(i,j)]; lab[i,j]=cur; cells=[]
                while stack:
                    x,y=stack.pop(); cells.append((x,y))
                    for dx in(-1,0,1):
                        for dy in(-1,0,1):
                            if dx==0 and dy==0: continue
                            nx,ny=x+dx,y+dy
                            if 0<=nx<H and 0<=ny<W and mask[nx,ny] and lab[nx,ny]==0:
                                lab[nx,ny]=cur; stack.append((nx,ny))
                comps.append(cells)
    return comps

def bbox(cells):
