from scipy import ndimage
def comps(g):
    g=np.array(g)
    mask=g!=8
    lab,n=ndimage.label(mask, structure=np.ones((3,3)))
    out=[]
    for i in range(1,n+1):
        ys,xs=np.where(lab==i)
        sub=g[ys.min():ys.max()+1, xs.min():xs.max()+1].copy()
        # blank outside? it's bbox so fine

def extract_pieces(g):
    g=np.array(g)
    lab,n=ndimage.label(g!=8, structure=np.ones((3,3)))
    pieces=[]
    for i in range(1,n+1):
        ys,xs=np.where(lab==i)
        sub=g[ys.min():ys.max()+1, xs.min():xs.max()+1]
        m=(lab[ys.min():ys.max()+1, xs.min():xs.max()+1]==i)
        m=m & (sub!=4)
        if m.sum()==0: continue
        # bounding box of remaining
        yy,xx=np.where(m)
        r0,r1,c0,c1=yy.min(),yy.max(),xx.min(),xx.max()
        piece=sub[r0:r1+1,c0:c1+1].copy()
        pieces.append((ys.min(),xs.min(),piece))
    return pieces

def firstfit(pieces,H,W,frame):
    grid=np.full((H,W),-1)
    for (r,c,pc) in pieces:
        ph,pw=pc.shape
        placed=False

                if np.all(grid[i:i+ph,j:j+pw]==-1):
                    grid[i:i+ph,j:j+pw]=pc
                    placed=True;break
            if placed:break
        if not placed: return None
    if np.any(grid==-1): return None
    return grid
