def analyze(g):

    H,W=g.shape

    return res

import numpy as np
def solve(g):
    g=np.array(g); H,W=g.shape
    # axis row = row with most 6s
    axis=int(np.argmax((g==6).sum(1)))
    lines=[]
    for c in range(W):
        rows=[r for r in range(H) if g[r,c] not in (8,6)]
        if rows:
            lines.append((c,g[rows[0],c],min(rows),max(rows)))
    cols=[l[0] for l in lines]
    lens=sorted([l[3]-l[2]+1 for l in lines])  # ascending
    out=np.full((H,W),8,dtype=int)
    # place axis row
    linecol={c:(col,len_) for (c,col,_,_),len_ in zip(lines,lens)}
    for c in range(W):
        out[axis,c]= linecol[c][0] if c in linecol else 6
    for c,(col,L) in linecol.items():
        half=L//2
        for r in range(axis-half,axis+half+1):
            out[r,c]=col
    return out
