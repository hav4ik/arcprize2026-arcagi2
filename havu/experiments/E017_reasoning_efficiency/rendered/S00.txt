import numpy as np

def extract_bars(a):
    H,W=a.shape
    counts=[np.sum(a[r]==6) for r in range(H)]
    axis=int(np.argmax(counts))
    cols={}
    for x in range(W):
        col=a[:,x]
        nz=[(y,col[y]) for y in range(H) if col[y] not in (8,) and col[y]!=6]
        if nz:
            color=nz[0][1]
            ys=[y for y,_ in nz]
            cols[x]=(color,min(ys),max(ys))
    # group consecutive columns with same color and same min/max
    bars=[]
    x=0
    xs=sorted(cols)
    i=0
    while i<len(xs):
        x0=xs[i]; color,ymin,ymax=cols[x0]
        j=i
        while j+1<len(xs) and xs[j+1]==xs[j]+1 and cols[xs[j+1]]==(color,ymin,ymax):
            j+=1
        x1=xs[j]
        bars.append((x0,x1,color,ymin,ymax))
        i=j+1
    return axis,bars
