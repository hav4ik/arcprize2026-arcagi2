import numpy as np
from scipy import ndimage

def comps(grid, colors):
    g=np.array(grid)
    for c in sorted(set(g.flatten().tolist())):
        if c==1: continue
        mask=(g==c)
        lab,n=ndimage.label(mask, structure=np.ones((3,3)))
        print("color",c,"components",n)
        for k in range(1,n+1):
            ys,xs=np.where(lab==k)

import numpy as np

def get_pieces(g,anchor,bg=1):
    g=np.array(g)
    pmask=(g!=bg)&(g!=anchor)
    ys,xs=np.where(pmask)
    r0,r1,c0,c1=ys.min(),ys.max(),xs.min(),xs.max()
    pat=g[r0:r1+1,c0:c1+1].copy()
    pat[~pmask[r0:r1+1,c0:c1+1]]=0
