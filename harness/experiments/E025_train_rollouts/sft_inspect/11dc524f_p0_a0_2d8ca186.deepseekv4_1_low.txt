def solve(g):
    A=np.array(g)
    H,W=A.shape
    red=[tuple(x) for x in np.argwhere(A==2)]
    grn=np.argwhere(A==5)
    rr0,rc0=np.min(red,axis=0); rr1,rc1=np.max(red,axis=0)
    gr0,gc0=grn.min(0); gr1,gc1=grn.max(0)
    # decide axis: which axis has a gap
    col_gap = (rc1 < gc0) or (rc0 > gc1)
    cells=[]
    if col_gap:
        if rc1 < gc0:  # red left -> move right so rc1 = gc0-1
            shift = (gc0-1)-rc1
        else:          # red right -> move left so rc0 = gc1+1
            shift = (gc1+1)-rc0
        for (r,c) in red:
            cells.append((r,c+shift))
        # contact line between cols; reflect c
        if rc1 < gc0:
            line2 = 2*gc0-1  # c' = line2 - c
        else:
            line2 = 2*gc1+1
        grn_cells=[(r,line2-c) for (r,c) in cells]
    else:
        if rr1 < gr0:  # red above
            shift=(gr0-1)-rr1
        else:          # red below
            shift=(gr1+1)-rr0
        for (r,c) in red:
            cells.append((r+shift,c))
        if rr1 < gr0:
            line2=2*gr0-1
        else:
            line2=2*gr1+1
        grn_cells=[(line2-r,c) for (r,c) in cells]
    out=np.full_like(A,7)
    for (r,c) in cells: out[r,c]=2
    for (r,c) in grn_cells: out[r,c]=5
    return out
