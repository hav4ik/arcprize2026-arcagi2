import numpy as np
from collections import deque

def get_comps(g, bg=8):
    g=np.array(g); H,W=g.shape
    res={}
    for c in sorted(set(g.flatten())-{bg}):
        seen=np.zeros_like(g,bool); out=[]
        for i in range(H):
            for j in range(W):
                if g[i,j]==c and not seen[i,j]:
                    q=deque([(i,j)]); seen[i,j]=True; cells=[]
                    while q:
                        a,b=q.popleft(); cells.append((a,b))
                        for da,db in((1,0),(-1,0),(0,1),(0,-1)):
                            x,y=a+da,b+db
                            if 0<=x<H and 0<=y<W and g[x,y]==c and not seen[x,y]:
                                seen[x,y]=True;q.append((x,y))
                    out.append(cells)
        res[c]=out
    return res

DIRS={'N':(-1,0),'S':(1,0),'E':(0,1),'W':(0,-1)}
CW={'N':'E','E':'S','S':'W','W':'N'}

def rot_cw_pt(p):
    r,c=p; return (c,-r)

def rot_shape(cells,k):
    pts=list(cells)
    for _ in range(k):
        pts=[rot_cw_pt(p) for p in pts]
    r0=min(p[0] for p in pts); c0=min(p[1] for p in pts)
    return [(a-r0,b-c0) for a,b in pts]

def rotate_dir(d,k):
    for _ in range(k):
        d=CW[d]
    return d

def tab_dirs(cells):
    rs=[p[0] for p in cells]; cs=[p[1] for p in cells]
    r0,r1,c0,c1=min(rs),max(rs),min(cs),max(cs)
    res=[]
    # N
    top=[p for p in cells if p[0]==r0]
    if len(top)==2 and ( (c1-c0+1)==2 or all(c0< p[1] <c1 for p in top) ): res.append('N')
    bot=[p for p in cells if p[0]==r1]
    if len(bot)==2 and ( (c1-c0+1)==2 or all(c0< p[1] <c1 for p in bot) ): res.append('S')
    left=[p for p in cells if p[1]==c0]
    if len(left)==2 and ( (r1-r0+1)==2 or all(r0< p[0] <r1 for p in left) ): res.append('W')
    right=[p for p in cells if p[1]==c1]
    if len(right)==2 and ( (r1-r0+1)==2 or all(r0< p[0] <r1 for p in right) ): res.append('E')

def tab_cells_in_dir(cells,d):
    rs=[p[0] for p in cells]; cs=[p[1] for p in cells]
    r0,r1,c0,c1=min(rs),max(rs),min(cs),max(cs)
    if d=='N': return sorted([p for p in cells if p[0]==r0])
    if d=='S': return sorted([p for p in cells if p[0]==r1])
    if d=='W': return sorted([p for p in cells if p[1]==c0])
    if d=='E': return sorted([p for p in cells if p[1]==c1])

def find_diamond(comps):
    # look for pattern: horizontal domino top at (r,c0),(r,c0+1); vertical left (r+1,c0-1),(r+2,c0-1); vertical right (r+1,c0+2),(r+2,c0+2); horizontal bottom (r+3,c0),(r+3,c0+1)
    H_dom=[]; V_dom=[]
    for c,lst in comps.items():
        for comp in lst:
            if len(comp)!=2: continue
            (r1_,c1_),(r2_,c2_)=sorted(comp)
            if r1_==r2_ and abs(c1_-c2_)==1: H_dom.append((c,min(c1_,c2_),r1_))  # (color, c0, r)
            if c1_==c2_ and abs(r1_-r2_)==1: V_dom.append((c,c1_,min(r1_,r2_)))
    for (ct,c0t,rt) in H_dom:
        # bottom
        for (cb,c0b,rb) in H_dom:
            if cb==ct: continue
            if c0b==c0t and rb==rt+3:
                # verticals
                L=[v for v in V_dom if v[1]==c0t-1 and v[2]==rt+1]
                R=[v for v in V_dom if v[1]==c0t+2 and v[2]==rt+1]
                if L and R:
                    return {'top':(ct,[(rt,c0t),(rt,c0t+1)]),
                            'bottom':(cb,[(rb,c0b),(rb,c0b+1)]),
                            'left':(L[0][0],[(rt+1,c0t-1),(rt+2,c0t-1)]),
                            'right':(R[0][0],[(rt+1,c0t+2),(rt+2,c0t+2)])}
    return None
