import numpy as np

def extract_panels(grid):
    g=np.array(grid)
    nr,nc=g.shape
    border=g[0,0]
    rowb=[all(g[i,:]==border) for i in range(nr)]
    colb=[all(g[:,j]==border) for j in range(nc)]
    # find row segments
    rows=[]
    i=0
    while i<nr:
        if not rowb[i]:
            j=i
            while j<nr and not rowb[j]: j+=1
            rows.append((i,j-1)); i=j
        else: i+=1
    cols=[]
    j=0
    while j<nc:
        if not colb[j]:
            k=j
            while k<nc and not colb[k]: k+=1
            cols.append((j,k-1)); j=k
        else: j+=1
    panels=[]
    for (r0,r1) in rows:
        for (c0,c1) in cols:
            panels.append(g[r0:r1+1,c0:c1+1])
    return panels, border

def shape_cells(panel):
    vals,counts=np.unique(panel,return_counts=True)
    bg=vals[np.argmax(counts)]
    cells=np.argwhere(panel!=bg)
    return bg,set(map(tuple,cells))
