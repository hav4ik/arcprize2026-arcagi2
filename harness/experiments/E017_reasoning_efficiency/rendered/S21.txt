def panels(inp):
    # separator columns: columns that are entirely the separator color (constant), where sep color != background
    h,w=inp.shape
    from collections import Counter
    cnt=Counter(inp.flatten())
    # background color = most common
    bg=cnt.most_common(1)[0][0]
    # find constant columns
    sep=[c for c in range(w) if len(set(inp[:,c]))==1 and inp[0,c]!=bg]
    # panel boundaries
    bounds=[]; start=0
    for c in sep:
        bounds.append((start,c)); start=c+1
    bounds.append((start,w))
    ps=[inp[:,a:b] for a,b in bounds]
    return bg,sep,ps

import numpy as np
from collections import Counter

def get_panels(inp):
    h,w=inp.shape
    cnt=Counter(inp.flatten())
    bg=cnt.most_common(1)[0][0]
    sep=[c for c in range(w) if len(set(inp[:,c]))==1 and inp[0,c]!=bg]
    bounds=[]; start=0
    for c in sep:
        bounds.append((start,c)); start=c+1
    bounds.append((start,w))
    return bg,[inp[:,a:b] for a,b in bounds]

def best_shift(m1,m2,rmax):
    best=None
    h,w=m1.shape
    for dr in range(-rmax,rmax+1):
        for dc in range(-rmax,rmax+1):
            s=np.roll(np.roll(m2,dr,0),dc,1)
            # only count where both defined (roll wraps; ignore borders)
            ov=(m1&s).sum()
            if best is None or ov>best[0]: best=(ov,dr,dc)
    return best

def best_align(a,b,rmax=8):
    # find shift of b so that it best matches a (overlap), no wrap
    best=None;h,w=a.shape
    for dr in range(-rmax,rmax+1):
        for dc in range(-rmax,rmax+1):
            s=np.zeros_like(b)
            # shift b by (dr,dc)
            for r in range(h):
                for c in range(w):
                    if b[r,c]:
                        nr,nc=r+dr,c+dc
                        if 0<=nr<h and 0<=nc<w: s[nr,nc]=1
            ov=(a&s).sum()
            tot=(a|s).sum()
            if best is None or ov>best[0] or (ov==best[0] and tot<best[2]): best=(ov, (dr,dc), tot)
    return best

def neighbors_count(m, diag=True):
    h,w=m.shape
    res=np.zeros_like(m,dtype=int)
    for r in range(h):
        for c in range(w):
            s=0
            for dr in (-1,0,1):
                for dc in (-1,0,1):
                    if dr==0 and dc==0: continue
                    if not diag and abs(dr)+abs(dc)!=2: continue
                    nr,nc=r+dr,c+dc
                    if 0<=nr<h and 0<=nc<w: s+=m[nr,nc]
            res[r,c]=s
    return res

def apply_rule(m,k,diag=True,alive_min=1):
    nb=neighbors_count(m,diag)
    return ((m==1)&(nb>=k)).astype(int)  # survival only

def shape_offsets(m):
    ys,xs=np.where(m==1)
    return set(zip((ys-ys.min()).tolist(),(xs-xs.min()).tolist()))

def transforms(offs):
    pts=list(offs); res=[]
    for _ in range(4):
        pts=[(c,-r) for r,c in pts]  # rotate 90
        res.append(frozenset((r-min(r0 for r0,_ in pts), c-min(c0 for _,c0 in pts)) for r,c in pts))
        res.append(frozenset((r,c*-1+max(c for _,c in pts)) for r,c in pts))
    # simpler: build via dihedral group
    base=list(offs)
    seen=set()
    out=[]
    import itertools
    for rot in range(4):
        p=base
        for _ in range(rot): p=[(c,-r) for r,c in p]
        for mir in (False,True):
            q=[(r,-c if mir else c) for r,c in p]
            mr=min(r for r,_ in q); mc=min(c for _,c in q)
            norm=frozenset((r-mr,c-mc) for r,c in q)
            if norm not in seen:
                seen.add(norm); out.append(norm)
    return out

def contains(mask, offs):
    h,w=mask.shape
    pts=list(offs)
    hh=max(r for r,c in pts); ww=max(c for r,c in pts)
    for r0 in range(h-hh):
        for c0 in range(w-ww):
            if all(mask[r0+r,c0+c]==1 for r,c in pts): return (r0,c0)
    return None

def transforms_list(offs):
    base=list(offs); seen=set(); out=[]
    for rot in range(4):
        p=base
        for _ in range(rot): p=[(c,-r) for r,c in p]
        for mir in (False,True):
            q=[(r,-c if mir else c) for r,c in p]
            mr=min(r for r,_ in q); mc=min(c for _,c in q)
            norm=frozenset((r-mr,c-mc) for r,c in q)
            if norm not in seen:
                seen.add(norm); out.append(norm)
    return out

def contains_all(mask, offs):

    return res
