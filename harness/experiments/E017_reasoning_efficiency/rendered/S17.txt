def solve(grid):
  g = np.array(grid)
  H,W = g.shape
  out = g.copy()
  # clear left region columns 0..8
  out[:, :9] = 0
  # collect colored cells in right region (cols 9..)
  colored = []
  for r in range(H):
    for c in range(9,W):
      v=g[r,c]
      if v!=0 and v!=5:
        colored.append((r,c,v))
        out[r,c]=0
  # shift up by 1
  for r,c,v in colored:
    if r-1>=1:
      out[r-1,c]=v
  # overlay shape
  for r in range(H):
    for c in range(0,9):
      if g[r,c]==5:
        if c+10 < W:
          out[r,c+10]=5
  return out
