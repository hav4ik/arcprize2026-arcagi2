from scipy import ndimage
def comps(g):
    g=np.array(g)
    res=[]
    for c in np.unique(g):
        if c==0: continue
        m=(g==c)
        lab,n=ndimage.label(m, structure=np.ones((3,3)))
        for i in range(1,n+1):
            ys,xs=np.where(lab==i)
            res.append((int(c),(int(ys.min()),int(xs.min()),int(ys.max()),int(xs.max())),len(ys)))
    return res

def markers(g):
    g=np.array(g); res=[]
    for c in np.unique(g):
        if c==0: continue
        m=(g==c); lab,n=ndimage.label(m,structure=np.ones((3,3)))
        for i in range(1,n+1):
            ys,xs=np.where(lab==i)
            if len(ys)==1: res.append((int(c),int(ys[0]),int(xs[0])))
    return res

def solve(g, verbose=False):
    g=np.array(g); H,W=g.shape
    out=np.zeros((H,W),int)
