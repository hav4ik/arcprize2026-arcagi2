import numpy as np
from collections import Counter, deque

def comps(cells):
    cells=set(cells); seen=set(); out=[]
    for c in cells:
        if c in seen: continue
        q=deque([c]); comp=[]; seen.add(c)
        while q:
            r,cc=q.popleft(); comp.append((r,cc))
            for dr,dc in((1,0),(-1,0),(0,1),(0,-1)):
                n=(r+dr,cc+dc)
                if n in cells and n not in seen:
                    seen.add(n); q.append(n)
        out.append(sorted(comp))
    return out

def corner(comp):
    # find cell adjacent to the other two
    s=set(comp)
    for c in comp:
        r,cc=c
        nb=sum(((r+dr,cc+dc) in s) for dr,dc in((1,0),(-1,0),(0,1),(0,-1)))
        if nb>=2: return c
    return None

def comps(cells):
    cells=set(cells); seen=set(); out=[]
    for c in cells:
        if c in seen: continue
        q=deque([c]); comp=[]; seen.add(c)
        while q:
            r,cc=q.popleft(); comp.append((r,cc))
            for dr,dc in((1,0),(-1,0),(0,1),(0,-1)):
                n=(r+dr,cc+dc)
                if n in cells and n not in seen: seen.add(n); q.append(n)
        out.append(sorted(comp))
    return out
def corner(comp):

        r,cc=c

        if nb>=2: return c
    return None

import numpy as np
from collections import Counter, deque

def comps(cells):
    cells=set(cells); seen=set(); out=[]
    for c in cells:
        if c in seen: continue
        q=deque([c]); comp=[]; seen.add(c)
        while q:
            r,cc=q.popleft(); comp.append((r,cc))
            for dr,dc in((1,0),(-1,0),(0,1),(0,-1)):
                n=(r+dr,cc+dc)
                if n in cells and n not in seen: seen.add(n); q.append(n)
        out.append(sorted(comp))
    return out
def corner(comp):
    s=set(comp)
    for c in comp:
        r,cc=c
        nb=sum(((r+dr,cc+dc) in s) for dr,dc in((1,0),(-1,0),(0,1),(0,-1)))
        if nb>=2: return c
    return None

def comps(cells):
    cells=set(cells); seen=set(); out=[]
    for c in cells:
        if c in seen: continue
        q=deque([c]); comp=[]; seen.add(c)
        while q:
            r,cc=q.popleft(); comp.append((r,cc))
            for dr,dc in((1,0),(-1,0),(0,1),(0,-1)):
                n=(r+dr,cc+dc)

        out.append(sorted(comp))
    return out
