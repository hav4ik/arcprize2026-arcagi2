def stamp(canvas,tile,mask,rr4,cc4):
    H,W=canvas.shape; out=canvas.copy()
    a,b=sorted(set(tile.flatten().tolist()))
    sw=np.where(tile==a,b,np.where(tile==b,a,tile)); bs=tile.shape[0]
    if (rr4,cc4)==(0,0): M=mask
    elif (rr4,cc4)==(0,2): M=np.rot90(mask,3)
    elif (rr4,cc4)==(2,2): M=np.rot90(mask,2)
    elif (rr4,cc4)==(2,0): M=np.rot90(mask,1)
    R,C=M.shape
    top=0 if rr4==0 else (H-R*bs if rr4==2 else (H-R*bs)//2)
    left=0 if cc4==0 else (W-C*bs if cc4==2 else (W-C*bs)//2)
    for r in range(R):
        for c in range(C):
            if M[r,c]:
                out[top+r*bs:top+r*bs+bs, left+c*bs:left+c*bs+bs]=sw
    return out
