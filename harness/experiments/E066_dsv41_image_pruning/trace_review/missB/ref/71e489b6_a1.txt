def noise_mask(i, w):
    # majority in w x w window (odd), noise if cell != majority
    pad=w//2
    ip=np.pad(i,pad,constant_values=-1)
    H,W=i.shape
    maj=np.zeros_like(i)
    for r in range(H):
        for c in range(W):
            win=ip[r:r+w,c:c+w].ravel()
            win=win[win>=0]
            maj[r,c]=1 if win.mean()>0.5 else 0
    return (i!=maj)

def solve(g):
    i=np.array(g).copy()
    H,W=i.shape
    other=np.zeros((H,W),int); own=np.zeros((H,W),int)
    for r in range(H):
        for c in range(W):
            win=i[max(0,r-1):r+2,max(0,c-1):c+2]
            own[r,c]=int((win==i[r,c]).sum())
            other[r,c]=win.size-own[r,c]
    minority=other>own
    c4=np.zeros((H,W),int)
    for r in range(H):
        for c in range(W):
            s=0
            for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                rr,cc=r+dr,c+dc
                if 0<=rr<H and 0<=cc<W and i[rr,cc]==i[r,c]: s+=1
            c4[r,c]=s
    defect=minority&(c4<=1)

                if 0<=rr<H and 0<=cc<W and not d0[rr,cc]:
