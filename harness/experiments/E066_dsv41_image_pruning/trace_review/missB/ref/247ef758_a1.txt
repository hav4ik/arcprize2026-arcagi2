from scipy.ndimage import label
import numpy as np

def solve(a, even_off='floor', priority='first'):
    a=np.array(a); H,W=a.shape
    # divider: find uniform nonzero column (prefer one followed by frame)
    divs=[c for c in range(W) if len(np.unique(a[:,c]))==1 and a[0,c]!=0]
    div=divs[0]
    fs=div+1
    frame=a[:,fs:]
    # base colors most common in top and left
    def mode(vals):
        u,c=np.unique(vals,return_counts=True)
        return u[np.argmax(c)]
    top=frame[0,:]; bot=frame[-1,:]; left=frame[:,0]; right=frame[:,-1]
    hb=mode(top); vb=mode(left)
    hpos={} ; vpos={}
    for j,v in enumerate(top):
        if v!=hb: hpos.setdefault(v,set()).add(j)
    for j,v in enumerate(bot):
        if v!=hb: hpos.setdefault(v,set()).add(j)
    for i,v in enumerate(left):
        if v!=vb: vpos.setdefault(v,set()).add(i)
    for i,v in enumerate(right):
        if v!=vb: vpos.setdefault(v,set()).add(i)
    # objects in left region 8-connectivity
    leftreg=a[:,:div]
    st=np.ones((3,3),bool)
    lab,n=label(leftreg!=0,structure=st)
    objs=[]
    for k in range(1,n+1):
        ys,xs=np.where(lab==k)
        col=leftreg[ys[0],xs[0]]
        # all same color assume
        r0,r1=ys.min(),ys.max(); c0,c1=xs.min(),xs.max()
        shape=np.zeros((r1-r0+1,c1-c0+1),int)
        shape[ys-r0,xs-c0]=leftreg[ys,xs]
        objs.append(dict(col=int(col),r0=r0,c0=c0,shape=shape,keep=False))
    out=a.copy()
    # determine used
    placed=[]
    for o in objs:
        c=o['col']
        if c in hpos and c in vpos:
            o['keep']=False
            # erase
            ys,xs=np.where(o['shape']!=0)
            out[o['r0']+ys,o['c0']+xs]=0
            placed.append(o)
        else:
            o['keep']=True
    # place
