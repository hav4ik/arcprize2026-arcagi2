import numpy as np, math

def seg_cells(a,b):
    r0,c0=a; r1,c1=b
    dr=r1-r0; dc=c1-c0
    if not (dr==0 or dc==0 or abs(dr)==abs(dc)):
        return None
    n=max(abs(dr),abs(dc))
    if n==0: return [a]
    out=[]
    for k in range(n+1):
        r=r0+round(dr*k/n); c=c0+round(dc*k/n)
        out.append((r,c))
    return out

def shape_multi(dots):
    dots=[tuple(d) for d in dots]
    if len(dots)==1: return set()
    pts=np.array(dots,float)
    cr,cc=pts.mean(axis=0)
    order=sorted(dots, key=lambda p: math.atan2(p[0]-cr, p[1]-cc))
    cells=set(dots)
    n=len(order)
    for i in range(n):
        a=order[i]; b=order[(i+1)%n]
        sc=seg_cells(a,b)
        if sc is None: continue
        cells.update(sc)
    return cells

def solve(grid):
    g=np.array(grid)
    H,W=g.shape
    colors=[c for c in np.unique(g) if c!=0]
    dots={c:[tuple(x) for x in np.argwhere(g==c)] for c in colors}
    multi=[c for c in colors if len(dots[c])>=2]
    shapes={c:shape_multi(dots[c]) for c in multi}
    out=np.zeros((H,W),int)
    # place dots
    for c in colors:
        for (r,cc) in dots[c]:
            out[r,cc]=c
    # draw multi shapes
    for c in multi:
        for (r,cc) in shapes[c]:
            if 0<=r<H and 0<=cc<W and out[r,cc]==0:
                out[r,cc]=c
    # singletons
    singles=[c for c in colors if len(dots[c])==1]
    for s in singles:
        (sr,sc)=dots[s][0]
        # nearest dot of another color that has multi shape
        best=None
        for p in multi:
            for (pr,pc) in dots[p]:
                d=abs(pr-sr)+abs(pc-sc)
                if best is None or d<best[0]:
                    best=(d,p,pr,pc)
        if best is None: continue
        d,p,pr,pc=best
        tr=sr-pr; tc=sc-pc
        for (r,cc) in shapes[p]:
            nr,nc=r+tr,cc+tc
            if 0<=nr<H and 0<=nc<W and out[nr,nc]==0:
                out[nr,nc]=s
    return out
