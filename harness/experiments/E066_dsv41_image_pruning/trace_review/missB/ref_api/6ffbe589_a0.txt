def rot_points(pts,k):
    # rotate set of points 90*k CCW about centroid of bbox, return set
    pts=np.array(pts)
    r0,r1=pts[:,0].min(),pts[:,0].max(); c0,c1=pts[:,1].min(),pts[:,1].max()
    # center (may be half-int)
    cy=(r0+r1)/2; cx=(c0+c1)/2
    out=set()
    for r,c in pts:
        y=r-cy; x=c-cx
        for _ in range(k):
            y,x=-x,y  # 90 CCW: (y,x)->(-x,y)? 
        out.add((round(y+cy),round(x+cx)))

def components(a):
    st=np.ones((3,3),int)
    lab,n=ndimage.label(a>0,st)
    comps=[]
    for j in range(1,n+1):
        ys,xs=np.where(lab==j)
        comps.append((j,ys,xs))
    return lab,n,comps

def analyze(a):
    lab,n,comps=components(a)
    sizes={j:len(ys) for j,ys,xs in comps}
    big=max(sizes,key=sizes.get)
    # dilation clustering
    dil=ndimage.binary_dilation(a>0,iterations=1)
    dl,dn=ndimage.label(dil,np.ones((3,3),int))
    # group containing big comp
    y,x=np.where(lab==big); grp=dl[y[0],x[0]]
    mainmask=np.zeros_like(a>0)
    ys,xs=np.where((lab>0)&(dl==grp))
    mainmask[ys,xs]=True
    # components in main vs legend
    main_comp_ids=set(lab[ys,xs])
    legend=[(j,ys,xs) for j,ys,xs in comps if j not in main_comp_ids]
    rr,cc=np.where(mainmask)
    r0,r1,c0,c1=rr.min(),rr.max(),cc.min(),cc.max()
    return a[r0:r1+1,c0:c1+1], mainmask[r0:r1+1,c0:c1+1], legend

def solve(a):
    a=np.array(a)
    crop,mainmask,legend=analyze(a)
    H,W=crop.shape
    n_by_color={}
    for j,ys,xs in legend:
        for col in np.unique(a[ys,xs]):
            n_by_color.setdefault(col,0)
            n_by_color[col]+=int((a[ys,xs]==col).sum())
    out=np.zeros((H,W),int)
    for col in np.unique(crop):
        if col==0: continue
        mask=(crop==col)
        k=0 if col not in n_by_color else (4-n_by_color[col])%4
        out[np.rot90(mask,k).astype(bool)]=col
    return out
