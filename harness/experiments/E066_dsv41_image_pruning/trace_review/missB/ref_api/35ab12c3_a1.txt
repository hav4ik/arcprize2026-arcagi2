import itertools, math
def line(a,b):
    r0,c0=a;r1,c1=b
    dr=r1-r0; dc=c1-c0
    if dr==0 or dc==0 or abs(dr)==abs(dc):
        n=max(abs(dr),abs(dc))
        sr=0 if dr==0 else dr//abs(dr); sc=0 if dc==0 else dc//abs(dc)
        return [(r0+sr*k,c0+sc*k) for k in range(n+1)]
    return None

def shape_of(dots):
    if len(dots)==1: return set(dots)
    cr=np.mean([d[0] for d in dots]); cc=np.mean([d[1] for d in dots])
    ds=sorted(dots,key=lambda d: math.atan2(d[0]-cr,d[1]-cc))
    sh=set(dots)
    n=len(ds)
    for i in range(n):
        L=line(ds[i],ds[(i+1)%n])
        if L: sh.update(L)
    return sh

def solve(g):
    g=np.array(g); H,W=g.shape
    cols=[int(c) for c in np.unique(g) if c!=0]
    dots={c: sorted(zip(*np.where(g==c))) for c in cols}
    prim=[c for c in cols if len(dots[c])>=2]
    sing=[c for c in cols if len(dots[c])==1]
    shapes={c:shape_of(dots[c]) for c in prim}
    out=np.zeros((H,W),dtype=int)
    # draw primaries
    for c in prim:
        for (r,cc) in shapes[c]: out[r,cc]=c
    for c in sing:
        d=dots[c][0]
        # nearest dot among primaries
        best=None
        for p in prim:
            for q in dots[p]:
                dist=(q[0]-d[0])**2+(q[1]-d[1])**2
                if best is None or dist<best[0]: best=(dist,p,q)
        p,q=best[1],best[2]
        vr=d[0]-q[0]; vc=d[1]-q[1]
        for (r,cc) in shapes[p]:
            nr,nc=r+vr,cc+vc
            if 0<=nr<H and 0<=nc<W and out[nr,nc]==0:
                out[nr,nc]=c
    return out
