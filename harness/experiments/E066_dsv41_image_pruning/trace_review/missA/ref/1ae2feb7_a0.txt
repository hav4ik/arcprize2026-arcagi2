import numpy as np

def find_divider(g):
    H,W=g.shape
    best=None
    for c in range(W):
        col=g[:,c]
        nz=col[col!=0]
        if nz.size>0 and len(set(nz.tolist()))==1 and nz.size>=H*0.5:
            # choose the one with max count
            if best is None or nz.size>best[1]:
                best=(c,nz.size,nz[0])
    return best[0] if best else None

def runs_in(seg):
    runs=[];c=0
    n=len(seg)
    while c<n:
        if seg[c]!=0:
            col=seg[c]; s=c
            while c<n and seg[c]==col: c+=1
            runs.append((s,c-s,col))
        else: c+=1
    return runs

def solve(g):
    g=np.array(g); o=g.copy()
    H,W=g.shape
    div=find_divider(g)
    if div is None:
        return o
    left=g[:,:div]; right=g[:,div+1:]
    left_has= np.any(left!=0)
    right_has=np.any(right!=0)
    rw=W-div-1
    if left_has and not right_has:
        # extend rightward
        for r in range(H):
            runs=runs_in(g[r,:div])
            if not runs: continue
            row=[0]*rw
            for s,L,col in runs:
                for i in range(rw):
                    if i % L ==0: row[i]=col
            for i in range(rw):
                o[r,div+1+i]=row[i]
    elif right_has and not left_has:

            for s,L,col in runs:
                for i in range(div):
                    if i % L==0: row[i]=col
            for i in range(div):
                o[r,div-1-i]=row[i]
    return o
