def solve(a):
    a=np.array(a)
    H,W=a.shape
    red=(a==2)
    out=a.copy()
    active=set()
    for y in range(H):
        for x in np.where(a[y]==6)[0]:
            active.add(int(x))
        for x in active:
            if not red[y,x]:
                out[y,x]=6
        if y==H-1: break
        newactive=set()
        for x in sorted(active):
            if red[y+1,x]:
                # red run in next row
                l=x
                while l-1>=0 and red[y+1,l-1]: l-=1
                r=x
                while r+1<W and red[y+1,r+1]: r+=1
                A=max(0,l-1); B=min(W-1,r+1)
                # reachable along current row through non-red cells
                lr=x
                while lr-1>=A and not red[y,lr-1]: lr-=1
                rr=x
                while rr+1<=B and not red[y,rr+1]: rr+=1
                for xx in range(lr,rr+1):
                    out[y,xx]=6
                # descend at reachable margins of red run (where below not red)
                if l-1>=0 and l-1>=lr and l-1<=rr:
                    newactive.add(l-1)
                if r+1<W and r+1>=lr and r+1<=rr:
                    newactive.add(r+1)
            else:
                newactive.add(x)
        active=newactive
    return out
