def find_divider(g):
    H,W=g.shape
    best=None
    for c in range(W):
        col=g[:,c]
        nz=set(int(x) for x in col if x!=0)
        if len(nz)==1:
            v=nz.pop()
            # count rows equal v
            cnt=sum(1 for x in col if x==v)
            if cnt>=H-2:
                return c,v
    return None,None

def blocks_in_row(row):
    res=[];s=None
    for c,v in enumerate(row):
        v=int(v)
        if v!=0 and s is None: s=[v,c]
        elif v==0 and s is not None:
            res.append((s[0],c-s[1])); s=None
        elif v!=0 and s is not None and v!=s[0]:
            res.append((s[0],c-s[1])); s=[v,c]
    if s is not None: res.append((s[0],len(row)-s[1]))
    return res

def solve(g):
    g=np.array(g); H,W=g.shape
    d,dv=find_divider(g)
    out=g.copy()
    for r in range(H):
        left=list(g[r,:d])
        bl=blocks_in_row(left)
        for (v,w) in bl:
            off=0
            while d+1+off<W:
                out[r,d+1+off]=v
                off+=w
    return out

def blocks_in_row(row):
    res=[];s=None
    for c,v in enumerate(row):
        v=int(v)
        if v!=0 and s is None: s=[v,c]
        elif v==0 and s is not None:
            res.append((s[0],c-s[1])); s=None
        elif v!=0 and s is not None and v!=s[0]:
            res.append((s[0],c-s[1])); s=[v,c]
    if s is not None: res.append((s[0],len(row)-s[1]))
    return res

def solve2(g):
    g=np.array(g); H,W=g.shape
    d,dv=find_divider(g)
    out=g.copy()
    for r in range(H):
        left=list(g[r,:d]); right=list(g[r,d+1:])
        bl=blocks_in_row(left)
        if bl:
            for (v,w) in bl:
                off=0
                while d+1+off<W:
                    out[r,d+1+off]=v; off+=w

                # draw farthest first -> reverse order, then nearest last

                    while d-1-off>=0:
                        out[r,d-1-off]=v; off+=w
    return out
