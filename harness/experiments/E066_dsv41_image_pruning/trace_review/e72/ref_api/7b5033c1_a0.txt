def solve(grid):

            comps.append({'color':int(c),'cells':set(zip(ys.tolist(),xs.tolist())),'ys':ys,'xs':xs,'count':len(ys),'minrow':ys.min()})
    # edges 8-adjacency between different comps
    adj={i:set() for i in range(len(comps))}
    for i in range(len(comps)):

            if comps[i]['color']==comps[j]['color']: continue
            found=False
            for (y,x) in comps[i]['cells']:
                for dy in (-1,0,1):
                    for dx in (-1,0,1):
                        if (y+dy,x+dx) in comps[j]['cells']:
                            found=True;break

    # find path starting topmost endpoint
