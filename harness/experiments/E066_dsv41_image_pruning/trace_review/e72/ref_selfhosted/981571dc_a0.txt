from collections import deque
N=30

def orbit(i,j):
    seen=set(); q=deque([(i,j)])
    while q:
        a,b=q.popleft()
        if (a,b) in seen: continue
        seen.add((a,b))
        # transpose
        cand=[(b,a)]
        if a>=2: cand.append((31-a,b))
        if b>=2: cand.append((a,31-b))
        for c in cand:
            if c not in seen: q.append(c)
    return seen

def solve_grid(inp):
    inp=np.array(inp)
    out=inp.copy()
    for i in range(N):
        for j in range(N):
            if out[i,j]==0:
                vals=[inp[x,y] for x,y in orbit(i,j) if inp[x,y]!=0]
                if vals:
                    # ensure all same
                    if len(set(vals))!=1:

                    out[i,j]=vals[0]
    return out
