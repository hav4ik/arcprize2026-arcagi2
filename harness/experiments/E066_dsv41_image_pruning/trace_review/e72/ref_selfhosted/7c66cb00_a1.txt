from scipy import ndimage
def comps(a, bg):
    mask = a!=bg
    lab,n=ndimage.label(mask, structure=np.ones((3,3)))
    out=[]
    for i in range(1,n+1):
        ys,xs=np.where(lab==i)
        sub=a[ys.min():ys.max()+1, xs.min():xs.max()+1]
        out.append((ys.min(),ys.max(),xs.min(),xs.max(),sub.copy()))
    return out

from scipy import ndimage

def solve(a):
    a=np.array(a).copy()
    H,W=a.shape
    bg=np.bincount(a.ravel()).argmax()
    mask=a!=bg
    lab,n=ndimage.label(mask, structure=np.ones((3,3)))
    band_mask=np.zeros_like(mask)
    bands=[]
    for i in range(1,n+1):
        ys,xs=np.where(lab==i)
        if xs.max()-xs.min()+1==W:
            band_mask[ys,xs]=True
            r0,r1=ys.min(),ys.max()
            corner=a[r0,0]
            inner=a[r0+1:r1,1:W-1]
            fill=np.bincount(inner.ravel()).argmax()
            bands.append((r0,r1,corner,fill))
    small=~band_mask & mask  # actually small objects all non-band colored cells
    out=a.copy()
    for (r0,r1,corner,fill) in bands:
        cells=np.argwhere(small & (a==fill))
        if len(cells)==0: continue
        maxr=cells[:,0].max()
        for (r,c) in cells:
            nr=r+(r1-maxr)
            if r0<=nr<=r1:
                out[nr,c]=corner
    return out
