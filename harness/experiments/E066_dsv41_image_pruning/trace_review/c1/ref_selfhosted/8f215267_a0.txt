def comps(g, bg):
    g=np.array(g); H,W=g.shape
    seen=np.zeros((H,W),bool); out=[]
    for i in range(H):

          col=g[i,j]; q=[(i,j)]; seen[i,j]=1; pts=[]

            x,y=q.pop(); pts.append((x,y))
