from scipy.ndimage import label

def solve(g):
    g=np.array(g)
    bg=np.bincount(g.ravel()).argmax()
    z=np.argwhere(g==0)
    r0,r1=z[:,0].min(),z[:,0].max()
    c0,c1=z[:,1].min(),z[:,1].max()
    H=r1-r0+1; W=c1-c0+1
    out=np.zeros((H,W),dtype=int)
    canvas=g[r0:r1+1,c0:c1+1]
    # background objects mask: non bg, non zero, outside canvas
    mask=(g!=bg)&(g!=0)
    mask[r0:r1+1,c0:c1+1]=False
    lab,n=label(mask, structure=np.ones((3,3)))
    # count components per color
    counts={}
    for i in range(1,n+1):
        color=g[lab==i][0]
        counts[color]=counts.get(color,0)+1
    for (rr,cc) in np.argwhere(canvas!=0):
        col=canvas[rr,cc]
        k=counts.get(col,0)
        for j in range(k):
            out[rr,cc+2*j]=col
    return out
