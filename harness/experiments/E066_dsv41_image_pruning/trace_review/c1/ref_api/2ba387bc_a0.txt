def shapes(g):
    g=np.array(g)
    seen=np.zeros_like(g,dtype=bool)
    out=[]
    for i in range(g.shape[0]):
        for j in range(g.shape[1]):
            if g[i,j]!=0 and not seen[i,j]:
                c=g[i,j]
                # flood fill same color
                stack=[(i,j)]; seen[i,j]=True; cells=[]
                while stack:
                    a,b=stack.pop(); cells.append((a,b))
                    for da,db in[(1,0),(-1,0),(0,1),(0,-1)]:
                        x,y=a+da,b+db
                        if 0<=x<g.shape[0] and 0<=y<g.shape[1] and not seen[x,y] and g[x,y]==c:
                            seen[x,y]=True; stack.append((x,y))
                rs=[p[0] for p in cells]; cs=[p[1] for p in cells]
                r0,r1=min(rs),max(rs); c0,c1=min(cs),max(cs)
                # classify hollow: interior all 0 and border full

                hollow = inner and (r1-r0+1)>2
                out.append((r0,c0,c,hollow))
    out.sort()
    return out
