def runs(grid):
    col=[row[0] for row in grid]
    res=[]
    for c in col:
        if res and res[-1][0]==c: res[-1][1]+=1
        else: res.append([c,1])
    return res

def solve(grid):
    r=runs(grid)
    H=len(grid)
    N=2*H-r[-1][1]
    out=[[0]*N for _ in range(N)]
    s=0
    for c,t in r[:-1]:
        for i in range(s,s+t):
            for j in range(s,N-s):
                out[i][j]=c; out[N-1-i][j]=c
                out[j][i]=c; out[j][N-1-i]=c
        s+=t
    lc,lt=r[-1]
    for i in range(s,N-s):
        for j in range(s,N-s):
            out[i][j]=lc
    return out
