import numpy as np
from scipy import ndimage

def objects(g):
    g=np.array(g)
    lab,n=ndimage.label(g>0)
    objs=[]
    for i in range(1,n+1):
        ys,xs=np.where(lab==i)
        c=int(g[ys[0],xs[0]])
        y0,y1,x0,x1=ys.min(),ys.max(),xs.min(),xs.max()
        h=y1-y0+1; w=x1-x0+1
        sub=g[y0:y1+1,x0:x1+1]
        # hollow if center 2x2 zero (for 4x4)
        hollow = (h==4 and w==4 and sub[1:3,1:3].sum()==0)
        objs.append((y0,x0,c,hollow,sub))
    objs.sort(key=lambda t:(t[0],t[1]))
    return objs

def solve(g):
