from scipy.ndimage import label

def get_components(a, col):
    m=a==col
    lab,n=label(m, structure=np.ones((3,3)))
    out=[]
    for k in range(1,n+1):
        pts=np.argwhere(lab==k)
        if len(pts)==0: continue
        r0,c0=pts.min(0); r1,c1=pts.max(0)
        out.append({'pts':pts,'size':len(pts),'bbox':(r0,c0,r1,c1),'area':(r1-r0+1)*(c1-c0+1)})
    return out
