def bands_of(inp):
    inp=np.array(inp); bands=[]
    for row in inp:
        c=int(row[0])
        if bands and bands[-1][0]==c: bands[-1][1]+=1
        else: bands.append([c,1])
    return bands

def solve(inp):
    inp=np.array(inp); T=inp.shape[0]
    bands=bands_of(inp)
    # colors by depth
    depth_col=[]
    for c,t in bands:
        depth_col += [c]*t
    n=2*(T-1)
    out=np.zeros((n,n),dtype=int)
    for r in range(n):
        for col in range(n):
            d=min(r,col,n-1-r,n-1-col)
            out[r,col]=depth_col[d] if d<len(depth_col) else depth_col[-1]
    return out
